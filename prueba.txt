    function calcularLineaCD(importe, tipo, porcentaje, incluirPorcentaje, hayCash) {
      if (tipo === "TC") return { base: importe, cargo: 0, total: importe, incluido: false };
      const cargo = Math.round(importe * porcentaje / 100);
      if (incluirPorcentaje) {
        if (hayCash) {
          const factor = 100 + porcentaje;
          // Mayor base en céntimos cuyo total truncado no supera el importe.
          const base = Math.floor(((importe + 1) * 100 - 1) / factor);
          const total = Math.floor(base * factor / 100);
          return { base, cargo: total - base, total, incluido: true };
        }
        const base = Math.round(importe * 100 / (100 + porcentaje));
        return { base, cargo: importe - base, total: importe, incluido: true };
      }
      return { base: importe, cargo, total: importe + cargo, incluido: false };
    }

        sumar(
          pagos.map((pago, i) =>
            pago / Math.pow(1 + tasa, diasAcumulados[i] / 365)
          )
        ) - monto;

      let inferior = -0.999999999;
      let superior = 1;

      while (valorPresente(superior) > 0 && superior < 1e16) {
        superior = superior * 2 + 1;
      }

      if (
        valorPresente(inferior) < 0 ||
        valorPresente(superior) > 0
      ) {
        return NaN;
      }

      for (let i = 0; i < 120; i++) {
        const medio = (inferior + superior) / 2;

        if (valorPresente(medio) > 0) {
          inferior = medio;
        } else {
          superior = medio;
        }
      }

      return Math.pow(
        1 + (inferior + superior) / 2,
        360 / 365
      ) - 1;
    }

    function simular(datos) {
      const {
        monto, plazo, tea, desembolso, primerPago,
        seguro, estadoCuenta, julio, diciembre, desempleo
      } = datos;

      const fechas = Array.from(
        { length: plazo },
        (_, i) => sumarMeses(primerPago, i)
      );

      const fechasAjustadas = fechas.map(ajustarDomingo);

      const dias = fechasAjustadas.map((fecha, i) =>
        diasEntre(
          fecha,
          i === 0 ? desembolso : fechasAjustadas[i - 1]
        )
      );

      const acumulados = fechasAjustadas.map(
        fecha => diasEntre(fecha, desembolso)
      );

      const factores = fechas.map(fecha => {
        const mes = fecha.getUTCMonth() + 1;
        return mes === 7 ? julio : mes === 12 ? diciembre : 1;
      });

      let primaPrincipal = 0;

      if (seguro === "Vida") {
        const duracion = acumulados[plazo - 1];
        const rango = PARAMETROS.vida.find(
          fila => duracion >= fila[0] && duracion <= fila[1]
        );

        if (!rango) {
          throw new Error(
            "La duración del préstamo queda fuera de las tarifas de vida disponibles."
          );
        }

        primaPrincipal = monto * rango[2] * duracion;
      } else if (
        Object.hasOwn(PARAMETROS.desgravamen, seguro)
      ) {
        const tarifa = PARAMETROS.desgravamen[seguro];
        const tem = Math.pow(1 + tea, 1 / 12) - 1;

        const cuotaAuxiliar = monto / sumar(
          factores.map((factor, i) =>
            factor / Math.pow(
              1 + tem + tarifa,
              acumulados[i] / 30
            )
          )
        );

        let saldoAuxiliar = monto;
        const primasMensuales = [];

        for (let i = 0; i < plazo; i++) {
          const interesAuxiliar =
            Math.pow(1 + tea, dias[i] / 360) - 1;

          const tasaPeriodoSeguro =
            Math.pow(1 + tarifa, dias[i] / 30) - 1;

          primasMensuales.push(
            tasaPeriodoSeguro * saldoAuxiliar
          );

          saldoAuxiliar = Math.max(
            0,
            saldoAuxiliar *
              (1 + interesAuxiliar + tasaPeriodoSeguro) -
              cuotaAuxiliar * factores[i]
          );
        }

        primaPrincipal = sumar(
          primasMensuales.map((prima, i) =>
            prima / Math.pow(
              1 + PARAMETROS.costoCapitalMensual,
              i + 1
            )
          )
        );
      }

      const adicionalDesempleo = redondearDos(
        (monto + primaPrincipal) * desempleo
      );

      const montoFinanciado =
        monto + primaPrincipal + adicionalDesempleo;

      const comision =
        estadoCuenta ? PARAMETROS.comisionFisica : 0;

      const cuotaBase = montoFinanciado / sumar(
        factores.map((factor, i) =>
          factor / Math.pow(1 + tea, acumulados[i] / 360)
        )
      );

      let saldo = montoFinanciado;
      const cronograma = [];
      const flujosTcea = [];

      for (let i = 0; i < plazo; i++) {
        const interes =
          saldo * (Math.pow(1 + tea, dias[i] / 360) - 1);

        const pagoSinComision = Math.min(
          saldo + interes,
          cuotaBase * factores[i]
        );

        const capital = Math.max(
          0,
          pagoSinComision - interes
        );

        const saldoFinal = Math.max(0, saldo - capital);

        cronograma.push({
          numero: i + 1,
          // Muestra la fecha ajustada que se usa para calcular intereses.
          fecha: fechasAjustadas[i],
          saldoInicial: saldo,
          interes,
          capital,
          comisiones: comision,
          cuota: pagoSinComision + comision,
          saldoFinal,
          tipo: factores[i] === 2 ? "Doble" : "Simple"
        });

        flujosTcea.push(cuotaBase * factores[i] + comision);
        saldo = saldoFinal;
      }

      return {
        cuotaEstimada: cuotaBase + comision,
        montoSolicitado: monto,
        primasSeguros: primaPrincipal + adicionalDesempleo,
        montoFinanciado,
        intereses: sumar(cronograma.map(fila => fila.interes)),
        comisiones: sumar(cronograma.map(fila => fila.comisiones)),
        totalPagar: sumar(cronograma.map(fila => fila.cuota)),
        tcea: calcularTcea(monto, flujosTcea, acumulados),
        adicionalDesempleo,
        cronograma
      };
    }

    function conectarSimulador(panel, buscar, obtenerMonto) {
      const $ = buscar;
      const formulario = $("formulario");
      const mensaje = $("mensaje");
      const cuerpoCronograma = $("cronograma");
      function leerDatos() {
        return {
          monto: obtenerMonto(),
          plazo: Number($("plazo").value),
          tea: Number($("tea").value) / 100,
          desembolso: leerFecha($("desembolso").value),
          primerPago: leerFecha($("primerPago").value),
          seguro: $("seguro").value,
          estadoCuenta: $("estadoCuenta").checked,
          julio: $("julio").checked ? 2 : 1,
          diciembre: $("diciembre").checked ? 2 : 1,
          desempleo: $("desempleo").checked ? 0.04 : 0
        };
      }

      function cambiarVisibilidad(abierto) {
        $("panelCronograma").hidden = !abierto;
        $("verCronograma").setAttribute("aria-expanded", String(abierto));
        $("textoBoton").textContent = abierto ? "Ocultar cronograma" : "Ver cronograma";
      }

      $("verCronograma").addEventListener("click", () => {
        cambiarVisibilidad($("panelCronograma").hidden);
      });

      function invalidarResultados() {
        cambiarVisibilidad(false);
        $("verCronograma").disabled = true;
        mensaje.textContent = "";

        panel.querySelectorAll("output").forEach(salida => {
          salida.textContent = "—";
        });

        cuerpoCronograma.replaceChildren();
        $("contenedorTabla").hidden = true;
        $("vacioCronograma").hidden = false;

        $("estadoResultado").textContent = "Sin calcular";

        $("notaCuota").textContent = "";

        $("detalleCronograma").textContent =
          "El detalle de tus cuotas aparecerá aquí.";
      }

      function mostrarCronograma(filas) {
        const fragmento = document.createDocumentFragment();

        filas.forEach(fila => {
          const tr = document.createElement("tr");

          const celdas = [
            fila.numero,
            mostrarFecha(fila.fecha),
            numero(fila.saldoInicial),
            numero(fila.interes),
            numero(fila.capital),
            numero(fila.comisiones),
            numero(fila.cuota),
            numero(fila.saldoFinal)
          ];

          celdas.forEach(texto => {
            const td = document.createElement("td");
            td.textContent = texto;
            tr.appendChild(td);
          });

          const tipo = document.createElement("td");

          if (fila.tipo === "Doble") {
            const badge = document.createElement("span");
            badge.className = "double-badge";
            badge.textContent = "Doble";
            tipo.appendChild(badge);
          } else {
            tipo.textContent = "Simple";
          }

          tr.appendChild(tipo);
          fragmento.appendChild(tr);
        });

        cuerpoCronograma.replaceChildren(fragmento);
        $("contenedorTabla").hidden = false;
        $("vacioCronograma").hidden = true;
      }

      function calcular() {
        invalidarResultados();

        validarFechaPrimerPago();
        if (!formulario.reportValidity()) return;

        const datos = leerDatos();

        if (
          !Number.isFinite(datos.monto) ||
          !Number.isFinite(datos.tea) ||
          !Number.isInteger(datos.plazo) ||
          datos.monto <= 0 ||
          datos.tea < 0 ||
          datos.plazo < 1 ||
          datos.plazo > 72 ||
          !Number.isFinite(datos.desembolso.getTime()) ||
          !Number.isFinite(datos.primerPago.getTime())
        ) {
          mensaje.textContent =
            "Revisa los importes, el plazo y las fechas ingresadas.";
          return;
        }


        try {
          const resultado = simular(datos);

          const resumenValido = Object.entries(resultado).every(
            ([clave, dato]) =>
              clave === "cronograma" || Number.isFinite(dato)
          );

          const cronogramaValido = resultado.cronograma.every(
            fila => [
              fila.saldoInicial,
              fila.interes,
              fila.capital,
              fila.comisiones,
              fila.cuota,
              fila.saldoFinal
            ].every(Number.isFinite)
          );

          if (!resumenValido || !cronogramaValido) {
            throw new Error(
              "No se pudo calcular con estos valores. Revisa los datos."
            );
          }

          Object.entries(resultado).forEach(([id, dato]) => {
            if (id !== "cronograma") {
              $(id).textContent = id === "tcea"
                ? numero(dato * 100) + "%"
                : moneda(dato);
            }
          });

          $("diasPrimerPago").textContent = String(diasEntre(datos.primerPago, datos.desembolso));
          mostrarCronograma(resultado.cronograma);
          $("verCronograma").disabled = false;

          $("estadoResultado").textContent = "Calculado";


          const hayDobles = resultado.cronograma.some(
            fila => fila.tipo === "Doble"
          );

          $("notaCuota").textContent = hayDobles
            ? "Las cuotas dobles figuran en el cronograma. La comisión por estado de cuenta no se duplica."
            : "";

          $("detalleCronograma").textContent =
            `${datos.plazo} cuotas · Importes en soles · ` +
            "Los vencimientos en domingo se trasladan al lunes.";


        } catch (error) {
          invalidarResultados();
          mensaje.textContent = error.message ||
            "Ocurrió un error al calcular. Revisa los datos.";
        }
      }

      formulario.addEventListener("submit", evento => {
        evento.preventDefault();
        calcular();
      });

      formulario.addEventListener("input", invalidarResultados);
      formulario.addEventListener("change", invalidarResultados);
      function configurarFechas(desembolso, primerPago, alerta) {
        function proponerPrimerPago() {
          const fecha = desembolso.valueAsDate;
          if (fecha) fecha.setUTCDate(fecha.getUTCDate() + 30);
          primerPago.valueAsDate = fecha;
        }

        function validar() {
          const inicio = desembolso.valueAsDate;
          const campo = primerPago;
          const fin = campo.valueAsDate;
          const dias = inicio && fin ? diasEntre(fin, inicio) : null;
          const fueraDeRango = dias !== null && (dias < 30 || dias > 60);
          const aviso = fueraDeRango
            ? "El primer pago debe ser entre 30 y 60 días calendario después del desembolso."
            : "";
          campo.setCustomValidity(aviso);
          campo.classList.toggle("is-invalid", fueraDeRango);
          campo.setAttribute("aria-invalid", String(fueraDeRango));
          alerta.textContent = aviso;
        }

        desembolso.addEventListener("input", () => {
          proponerPrimerPago();
          validar();
        });
        primerPago.addEventListener("input", validar);
        return validar;
      }

      const validarFechaPrimerPago = configurarFechas($("desembolso"), $("primerPago"), $("avisoPrimerPago"));
      formulario.addEventListener("reset", () => {
        invalidarResultados();
        queueMicrotask(validarFechaPrimerPago);
      });
      invalidarResultados();
    }

    conectarSimulador($("productoPP"), $, () => Number($("monto").value));
    const buscarCD = id => $("cd" + id[0].toUpperCase() + id.slice(1));
    conectarSimulador($("productoCD"), buscarCD, actualizarResumenCD);
    $("cdFormulario").addEventListener("reset", () => queueMicrotask(() => {
      filasCD.replaceChildren(nuevaFilaCD());
      actualizarFilasCD();
      actualizarResumenCD();
    }));

    const pestanas = [$("tabPP"), $("tabCD")];
    function seleccionarPestana(activa) {
      for (const pestana of pestanas) {
        const seleccionada = pestana === activa;
        pestana.setAttribute("aria-selected", String(seleccionada));
        pestana.tabIndex = seleccionada ? 0 : -1;
        $(pestana.getAttribute("aria-controls")).hidden = !seleccionada;
      }
    }

    pestanas.forEach((pestana, indice) => {
      pestana.addEventListener("click", () => seleccionarPestana(pestana));
      pestana.addEventListener("keydown", evento => {
        const destinos = { ArrowLeft: 1 - indice, ArrowRight: 1 - indice, Home: 0, End: 1 };
        if (!Object.hasOwn(destinos, evento.key)) return;
        evento.preventDefault();
        const destino = pestanas[destinos[evento.key]];
        seleccionarPestana(destino);
        destino.focus();
      });
    });

  </script>
