  <script>
    const formulario = document.getElementById('formulario');
    const cuotas = document.getElementById('cuotas');
    const resultados = document.getElementById('resultados');

    const MILISEGUNDOS_POR_DIA = 24 * 60 * 60 * 1000;

    function formatearFecha(fecha) {
      const dia = String(fecha.getUTCDate()).padStart(2, '0');
      const mes = String(fecha.getUTCMonth() + 1).padStart(2, '0');
      const anio = String(fecha.getUTCFullYear()).padStart(4, '0');

      return `${dia}/${mes}/${anio}`;
    }

    function diferenciaDias(fechaFinal, fechaInicial) {
      return (
        (fechaFinal.getTime() - fechaInicial.getTime()) /
        MILISEGUNDOS_POR_DIA
      );
    }

    function redondear(valor, decimales) {
      const [base, exponente = '0'] =
        Math.abs(valor).toString().split('e');

      const desplazado = Number(
        base + 'e' + (Number(exponente) + decimales)
      );

      return (
        Math.sign(valor) *
        Math.round(desplazado) /
        (10 ** decimales)
      );
    }

    function formatearPorcentaje(valor) {
      if (!Number.isFinite(valor)) {
        return 'No calculable';
      }

      return redondear(valor * 100, 2).toFixed(2) + '%';
    }

    /*
      TIR para un importe inicial negativo seguido de cuotas
      no negativas, considerando cada cuota como un período mensual.

      La estimación inicial será la TEM.
      Se busca la tasa que hace que el valor actual de los flujos sea 0.
    */
    function calcularTIR(flujos, estimacion) {
      if (
        flujos.length < 2 ||
        !flujos.every(Number.isFinite) ||
        flujos[0] >= 0 ||
        flujos.slice(1).some(valor => valor < 0) ||
        !flujos.slice(1).some(valor => valor > 0)
      ) {
        return null;
      }

      // Normalizar los flujos para mejorar la precisión numérica.
      const escala = -flujos[0];
      const normalizados = flujos.map(valor => valor / escala);

      function valorActual(tasa) {
        let valor = normalizados[normalizados.length - 1];

        for (let i = normalizados.length - 2; i >= 0; i--) {
          valor = normalizados[i] + valor / (1 + tasa);
        }

        return valor;
      }

      let inferior = -1;
      let superior = Math.max(1, estimacion);
      let valorSuperior = valorActual(superior);

      // Ampliar el intervalo si la tasa buscada es mayor.
      for (let i = 0; i < 100 && valorSuperior > 0; i++) {
        superior = superior * 2 + 1;
        valorSuperior = valorActual(superior);
      }

      if (
        !Number.isFinite(superior) ||
        Number.isNaN(valorSuperior) ||
        valorSuperior > 0
      ) {
        return null;
      }

      // Primera evaluación usando la estimación TEM.
      let tasa = Math.max(0, Math.min(estimacion, superior));

      for (let i = 0; i < 200; i++) {
        const valor = valorActual(tasa);

        if (Number.isNaN(valor)) {
          return null;
        }

        if (Math.abs(valor) <= 1e-12) {
          return tasa;
        }

        if (valor > 0) {
          inferior = tasa;
        } else {
          superior = tasa;
        }

        const siguiente = (inferior + superior) / 2;

        if (siguiente === inferior || siguiente === superior) {
          return tasa;
        }

        tasa = siguiente;
      }

      return null;
    }

    formulario.addEventListener('submit', function (evento) {
      evento.preventDefault();

      const importe = document.getElementById('importe').valueAsNumber;
      const plazo = document.getElementById('plazo').valueAsNumber;
      const ciclo = Number(document.getElementById('ciclo').value);

      // Ejemplo: ingresar 10 equivale a una TEA del 10 %.
      const tea = document.getElementById('tea').valueAsNumber / 100;

      // 0.40 % = 0.004; 0.30 % = 0.003.
      const seguro = Number(
        document.getElementById('seguro').value
      ) / 100;

      const fechaIngresada = document.getElementById('fecha').value;
      const fechaDesembolso = new Date(
        fechaIngresada + 'T00:00:00Z'
      );

      // Primera facturación:
      // día menor o igual al ciclo: mismo mes;
      // día mayor al ciclo: mes siguiente.
      const mesesAdicionales =
        fechaDesembolso.getUTCDate() <= ciclo ? 0 : 1;

      const mesPrimeraCuota =
        fechaDesembolso.getUTCMonth() + mesesAdicionales;

      const cronograma = [];
      let vencimientoAnterior = null;
      let diasAcumulados = 0;

      cuotas.replaceChildren();
      resultados.replaceChildren();

      // Calcular fechas, días y factores.
      for (let numero = 1; numero <= plazo; numero++) {
        const fechaFacturacion = new Date(fechaDesembolso);

        fechaFacturacion.setUTCDate(ciclo);
        fechaFacturacion.setUTCMonth(
          mesPrimeraCuota + numero - 1
        );

        // Vencim: restar 3 días y luego sumar 1 mes.
        const fechaVencim = new Date(fechaFacturacion);

        fechaVencim.setUTCDate(
          fechaVencim.getUTCDate() - 3
        );

        fechaVencim.setUTCMonth(
          fechaVencim.getUTCMonth() + 1
        );

        let dias;

        if (numero === 1) {
          dias = diferenciaDias(
            fechaVencim,
            fechaDesembolso
          ) + 1;
        } else {
          dias = diferenciaDias(
            fechaVencim,
            vencimientoAnterior
          );
        }

        diasAcumulados += dias;

        // Factor redondeado a 9 decimales.
        const factorSinRedondear =
          1 / Math.pow(1 + tea, diasAcumulados / 360);

        const factor =
          Math.round(factorSinRedondear * 1e9) / 1e9;

        cronograma.push({
          numero,
          fechaFacturacion,
          fechaVencim,
          dias,
          diasAcumulados,
          factor
        });

        vencimientoAnterior = fechaVencim;
      }

      // Suma de los factores, sin redondear adicionalmente la suma.
      const sumaFactor = cronograma.reduce(function (suma, dato) {
        return suma + dato.factor;
      }, 0);

      console.log(
        'Suma de la columna Factor (sin redondeo adicional):',
        sumaFactor
      );

      // CuotaY: importe / suma de Factor, a 2 decimales.
      const cuotaY = redondear(importe / sumaFactor, 2);

      // Primer flujo de la TIR: importe ingresado en negativo.
      const flujos = [-importe];

      let principalPendiente = importe;
      let amortizacionAcumulada = 0;
      let amortizacionPrincipalAcumulada = 0;

      cronograma.forEach(function (dato, indice) {
        const esUltimaFila = indice === cronograma.length - 1;

        let amortizacion;
        let interesY;

        if (esUltimaFila) {
          amortizacion = principalPendiente;
          interesY = redondear(cuotaY - amortizacion, 2);
        } else {
          interesY = principalPendiente * (
            Math.pow(1 + tea, dato.dias / 360) - 1
          );

          amortizacion = cuotaY - interesY;
        }

        // Redondear los importes antes de calcular el saldo.
        principalPendiente = redondear(principalPendiente, 2);
        amortizacion = redondear(amortizacion, 2);

        // Saldo final: Principal pendiente - Amortización.
        const saldoFinal = redondear(
          principalPendiente - amortizacion,
          2
        );

        // Acumulado de Amortización, incluida la fila actual.
        amortizacionAcumulada += amortizacion;

        // Amortización Principal.
        const amortizacionPrincipal =
          amortizacionAcumulada <= 0
            ? 1
            : amortizacionAcumulada - amortizacionPrincipalAcumulada;

        // Cuota: copiar CuotaY.
        const cuota = cuotaY;

        // Interés: Cuota - Amortización Principal.
        const interes = cuota - amortizacionPrincipal;

        // Seguro de desgravamen con máximo de 30.00.
        const segDesgravamen = redondear(
          Math.min(principalPendiente * seguro, 30),
          2
        );

        // Cuota + Seguro de desgravamen.
        const cuotaConSeguro = redondear(
          cuota + segDesgravamen,
          2
        );

        // Agregar cada cuota con seguro a los flujos de la TIR.
        flujos.push(cuotaConSeguro);

        const vencimTexto = formatearFecha(dato.fechaVencim);

        const datosFila = [
          dato.numero,
          formatearFecha(dato.fechaFacturacion),
          vencimTexto,
          dato.dias,
          dato.diasAcumulados,
          dato.factor.toFixed(9),
          principalPendiente.toFixed(2),
          amortizacion.toFixed(2),
          interesY.toFixed(2),
          cuotaY.toFixed(2),
          saldoFinal.toFixed(2),
          amortizacionPrincipal.toFixed(2),
          interes.toFixed(2),
          cuota.toFixed(2),
          segDesgravamen.toFixed(2),
          cuotaConSeguro.toFixed(2),
          vencimTexto
        ];

        const fila = cuotas.insertRow();

        datosFila.forEach(function (valor) {
          fila.insertCell().textContent = valor;
        });

        principalPendiente = saldoFinal;
        amortizacionPrincipalAcumulada += amortizacionPrincipal;
      });

      // TEM = (1 + TEA)^(1/12) - 1.
      const tem = Math.pow(1 + tea, 1 / 12) - 1;

      // TIR usando TEM como estimación inicial.
      const tirMensual = calcularTIR(flujos, tem);

      // TCEA = (1 + TIR)^12 - 1.
      const tcea = tirMensual === null
        ? null
        : Math.pow(1 + tirMensual, 12) - 1;

      // TCEM = (1 + TCEA)^(1/12) - 1.
      const tcem = tcea === null
        ? null
        : Math.pow(1 + tcea, 1 / 12) - 1;

      resultados.innerHTML = `
        <p><strong>TEM:</strong> ${formatearPorcentaje(tem)}</p>
        <p><strong>TCEA:</strong> ${formatearPorcentaje(tcea)}</p>
        <p><strong>TCEM:</strong> ${formatearPorcentaje(tcem)}</p>
      `;
    });

    formulario.addEventListener('reset', function () {
      cuotas.replaceChildren();
      resultados.replaceChildren();
    });
  </script>
