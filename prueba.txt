  <script>
  'use strict';

  // Fechas en UTC: cada diferencia representa días de calendario completos.
  const MILISEGUNDOS_POR_DIA = 24 * 60 * 60 * 1000;

  function formatearFecha(fecha) {
    const dia = String(fecha.getUTCDate()).padStart(2, '0');
    const mes = String(fecha.getUTCMonth() + 1).padStart(2, '0');
    const anio = String(fecha.getUTCFullYear()).padStart(4, '0');
    return dia + '/' + mes + '/' + anio;
  }

  function diferenciaDias(fechaFinal, fechaInicial) {
    return (fechaFinal.getTime() - fechaInicial.getTime()) / MILISEGUNDOS_POR_DIA;
  }

  function redondear(valor, decimales) {
    const [base, exponente = '0'] = Math.abs(valor).toString().split('e');
    const desplazado = Number(base + 'e' + (Number(exponente) + decimales));
    return Math.sign(valor) * Math.round(desplazado) / (10 ** decimales);
  }

  function formatearPorcentaje(valor) {
    if (!Number.isFinite(valor)) return 'No calculable';
    return redondear(valor * 100, 2).toFixed(2) + '%';
  }

  // TIR de un desembolso negativo seguido de cuotas mensuales no negativas.
  // Los flujos se normalizan por el importe inicial y se resuelven por bisección.
  function calcularTIR(flujos, estimacion) {
    if (
      flujos.length < 2 ||
      !flujos.every(Number.isFinite) ||
      flujos[0] >= 0 ||
      flujos.slice(1).some(valor => valor < 0) ||
      !flujos.slice(1).some(valor => valor > 0)
    ) {
      return null;
    }

    const escala = -flujos[0];
    const normalizados = flujos.map(valor => valor / escala);

    function valorActual(tasa) {
      let valor = normalizados[normalizados.length - 1];
      for (let i = normalizados.length - 2; i >= 0; i--) {
        valor = normalizados[i] + valor / (1 + tasa);
      }
      return valor;
    }

    let inferior = -1;
    let superior = Math.max(1, estimacion);
    let valorSuperior = valorActual(superior);
    for (let i = 0; i < 100 && valorSuperior > 0; i++) {
      superior = superior * 2 + 1;
      valorSuperior = valorActual(superior);
    }
    if (!Number.isFinite(superior) || Number.isNaN(valorSuperior) || valorSuperior > 0) {
      return null;
    }

    let tasa = Math.max(0, Math.min(estimacion, superior));
    for (let i = 0; i < 200; i++) {
      const valor = valorActual(tasa);
      if (Number.isNaN(valor)) return null;
      if (Math.abs(valor) <= 1e-12) return tasa;

      if (valor > 0) {
        inferior = tasa;
      } else {
        superior = tasa;
      }
      const siguiente = (inferior + superior) / 2;
      if (siguiente === inferior || siguiente === superior) return tasa;
      tasa = siguiente;
    }
    return null;
  }

  // Fechas, días y factores de descuento de las cuotas, en el orden del modelo original.
  function calcularPeriodos(fechaDesembolso, ciclo, plazo, tea) {
    const mesesAdicionales = fechaDesembolso.getUTCDate() <= ciclo ? 0 : 1;
    const mesPrimeraCuota = fechaDesembolso.getUTCMonth() + mesesAdicionales;
    const periodos = [];
    let vencimientoAnterior = null;
    let diasAcumulados = 0;

    for (let numero = 1; numero <= plazo; numero++) {
      const fechaFacturacion = new Date(fechaDesembolso);
      fechaFacturacion.setUTCDate(ciclo);
      fechaFacturacion.setUTCMonth(mesPrimeraCuota + numero - 1);

      // Al ciclo se le restan tres días y luego se suma un mes.
      const fechaVencim = new Date(fechaFacturacion);
      fechaVencim.setUTCDate(fechaVencim.getUTCDate() - 3);
      fechaVencim.setUTCMonth(fechaVencim.getUTCMonth() + 1);

      const dias = numero === 1
        ? diferenciaDias(fechaVencim, fechaDesembolso) + 1
        : diferenciaDias(fechaVencim, vencimientoAnterior);
      diasAcumulados += dias;
      const factorSinRedondear = 1 / Math.pow(1 + tea, diasAcumulados / 360);
      const factor = Math.round(factorSinRedondear * 1e9) / 1e9;
      periodos.push({ numero, fechaFacturacion, fechaVencim, dias, diasAcumulados, factor });
      vencimientoAnterior = fechaVencim;
    }
    return periodos;
  }

  function calcularSimulacion({ importe, plazo, ciclo, fechaDesembolso, tea, seguro }) {
    const cronograma = calcularPeriodos(fechaDesembolso, ciclo, plazo, tea);
    // Se redondea cada factor a nueve decimales antes de sumar la columna.
    const sumaFactor = cronograma.reduce((suma, dato) => suma + dato.factor, 0);
    const cuotaY = redondear(importe / sumaFactor, 2);
    const flujos = [-importe];
    let principalPendiente = importe;
    let amortizacionAcumulada = 0;
    let amortizacionPrincipalAcumulada = 0;

    cronograma.forEach(function (dato, indice) {
      let amortizacion;
      let interesY;
      if (indice === cronograma.length - 1) {
        amortizacion = principalPendiente;
        interesY = redondear(cuotaY - amortizacion, 2);
      } else {
        interesY = principalPendiente * (Math.pow(1 + tea, dato.dias / 360) - 1);
        amortizacion = cuotaY - interesY;
      }

      principalPendiente = redondear(principalPendiente, 2);
      amortizacion = redondear(amortizacion, 2);
      const saldoFinal = redondear(principalPendiente - amortizacion, 2);
      amortizacionAcumulada += amortizacion;

      // Esta regla distingue la amortización principal de la amortización de la columna previa.
      const amortizacionPrincipal = amortizacionAcumulada <= 0
        ? 1
        : amortizacionAcumulada - amortizacionPrincipalAcumulada;
      const cuota = cuotaY;
      const interes = cuota - amortizacionPrincipal;
      const segDesgravamen = redondear(Math.min(principalPendiente * seguro, 30), 2);
      const cuotaConSeguro = redondear(cuota + segDesgravamen, 2);
      flujos.push(cuotaConSeguro);

      // Se conserva el principal de esta fila antes de pasar al saldo del mes siguiente.
      Object.assign(dato, {
        principalPendiente, amortizacion, interesY, cuotaY, saldoFinal,
        amortizacionPrincipal, interes, cuota, segDesgravamen, cuotaConSeguro
      });
      principalPendiente = saldoFinal;
      amortizacionPrincipalAcumulada += amortizacionPrincipal;
    });

    // El interés usa días/360; la TIR toma cada pago como un período mensual.
    const tem = Math.pow(1 + tea, 1 / 12) - 1;
    const tirMensual = calcularTIR(flujos, tem);
    const tcea = tirMensual === null ? null : Math.pow(1 + tirMensual, 12) - 1;
    const tcem = tcea === null ? null : Math.pow(1 + tcea, 1 / 12) - 1;
    return { cronograma, tem, tcea, tcem };
  }

  const importesPrimeraCuota = {

  const calendarioPeru = new Intl.DateTimeFormat('es-PE', {

  function hoyEnPeru(instante = new Date()) {

  }

  function marcarError(control, hayError) {

  }

  formulario.addEventListener('submit', function (evento) {

  formulario.addEventListener('change', alEditar);

    campoFecha.defaultValue = hoyEnPeru();
    for (const control of controles) marcarError(control, false);
    limpiarResultados();
    mostrarCronograma(false);
    document.getElementById('vista-pagos').checked = true;
    seccionCronograma.dataset.vista = 'pagos';
    contenedorTabla.scrollLeft = 0;
    aviso.textContent = 'Formulario y resultados limpios.';
  });

  for (const opcion of document.querySelectorAll('input[name="vista"]')) {
    opcion.addEventListener('change', function () {
      seccionCronograma.dataset.vista = opcion.value;
      contenedorTabla.scrollLeft = 0;
      actualizarNotaTabla();
    });
  }
  if ('ResizeObserver' in window) {
    new ResizeObserver(actualizarNotaTabla).observe(contenedorTabla);
  } else {
    window.addEventListener('resize', actualizarNotaTabla);
  }
