    function simular() {
      const tipoSeguro = document.getElementById("seguro").value;

      // Valores expresados en porcentaje.
      const primas = {
        "vida": 3.7000,
        "desgravamen": 0.1995,
        "desgravamen-devolucion": 0.2594,
        "sin-seguro": 0,
        "endosado": 0
      };

      const seguroPrima = primas[tipoSeguro];
      const costoCapital = 0.2871;

      console.log(
        "Seguro prima:",
        seguroPrima === 0 ? "0%" : seguroPrima.toFixed(4) + "%"
      );

      console.log(
        "Costo capital:",
        costoCapital.toFixed(4) + "%"
      );

      const campoTea = document.getElementById("tea");
      const campoImporte = document.getElementById("importe");
      const campoPlazo = document.getElementById("plazo");
      const campoFecha = document.getElementById("primer-pago");
      const campoDesembolso = document.getElementById("desembolso");

      if (!campoTea.reportValidity()) return;
      if (!campoImporte.reportValidity()) return;
      if (!campoPlazo.reportValidity()) return;
      if (!campoFecha.reportValidity()) return;
      if (!campoDesembolso.reportValidity()) return;

      // Convertimos los porcentajes a tasas.
      const tea = Number(campoTea.value) / 100;
      const tasaSeguroPrima = seguroPrima / 100;
      const tasaCostoCapital = costoCapital / 100;

      // TEA total prima.
      const teaTotalPrima = Math.pow(
        1 + (
          (Math.pow(1 + tea, 30 / 360) - 1)
          + tasaSeguroPrima
        ),
        12
      ) - 1;

      console.log(
        "TEA total prima:",
        (teaTotalPrima * 100) + "%"
      );

      const importe = Number(campoImporte.value);
      const cantidadCuotas = Number(campoPlazo.value);

      const cuotaJulioActiva =
        document.getElementById("cuota-julio").checked;

      const cuotaDiciembreActiva =
        document.getElementById("cuota-diciembre").checked;

      // Seguro de desempleo: 4% si está marcado; de lo contrario, 0%.
      const tasaSeguroDesempleo =
        document.getElementById("seguro-desempleo").checked
          ? 0.04
          : 0;

      const fechaDesembolso = new Date(
        campoDesembolso.value + "T00:00:00"
      );

      let fechaPago = new Date(
        campoFecha.value + "T00:00:00"
      );

      let fechaAnterior = fechaDesembolso;
      let diasAcumulados = 0;

      let sumaFactor1 = 0;
      let sumaFactor2 = 0;

      const cronograma = [];

      // PRIMER RECORRIDO:
      // Fechas, días y factores.
      for (let cuota = 1; cuota <= cantidadCuotas; cuota++) {
        const fechaPagoPrincipal = new Date(fechaPago);

        // Si cae domingo, pasa al lunes siguiente.
        if (fechaPagoPrincipal.getDay() === 0) {
          fechaPagoPrincipal.setDate(
            fechaPagoPrincipal.getDate() + 1
          );
        }

        const dias = calcularDias(
          fechaAnterior,
          fechaPagoPrincipal
        );

        diasAcumulados += dias;

        const mesPagoPrincipal =
          fechaPagoPrincipal.getMonth() + 1;

        const cuotaDoble =
          (mesPagoPrincipal === 7 && cuotaJulioActiva) ||
          (mesPagoPrincipal === 12 && cuotaDiciembreActiva);

        const denominadorFactor = Math.pow(
          1 + teaTotalPrima,
          diasAcumulados / 360
        );

        const factor1 =
          (cuotaDoble ? 2 : 1) / denominadorFactor;

        const factor2 = 1 / denominadorFactor;

        sumaFactor1 += factor1;
        sumaFactor2 += factor2;

        cronograma.push({
          numero: cuota,
          fechaPago: new Date(fechaPago),
          fechaPagoPrincipal,
          dias,
          diasAcumulados,
          cuotaDoble,
          factor1,
          factor2
        });

        fechaAnterior = fechaPagoPrincipal;
        fechaPago = sumarUnMes(fechaPago);
      }

      // Cuota prima.
      const cuotaPrima = importe / sumaFactor1;

      console.log("Suma Factor1:", sumaFactor1);
      console.log("Suma Factor2:", sumaFactor2);
      console.log("Cuota prima:", cuotaPrima);

      let saldoInicial = importe;
      let filas = "";
      let VPN = 0;

      // SEGUNDO RECORRIDO:
      // Cuotas, saldos y VPN.
      for (const fila of cronograma) {
        // Interés.
        const interes = (
          Math.pow(1 + tea, fila.dias / 360) - 1
        ) * saldoInicial;

        // Seguro de desgravamen.
        const segDesgravamen = (
          Math.pow(1 + tasaSeguroPrima, fila.dias / 30) - 1
        ) * saldoInicial;

        // Cuota normal.
        let cuotaConstante = cuotaPrima;

        // Cuota doble según el mes y los checks.
        if (fila.cuotaDoble) {
          cuotaConstante = 2 * cuotaPrima;
        }

        // La última cuota tiene prioridad.
        if (fila.numero === cantidadCuotas) {
          cuotaConstante =
            saldoInicial + interes + segDesgravamen;
        }

        // Amortización.
        const amortizacion =
          cuotaConstante - segDesgravamen - interes;

        // Saldo final.
        const saldoFinal = saldoInicial - amortizacion;

        // Equivalente a VNA(costo capital; seguros).
        // La primera fila se descuenta un período.
        VPN += segDesgravamen / Math.pow(
          1 + tasaCostoCapital,
          fila.numero
        );

        // Dos decimales únicamente para mostrar la tabla.
        filas += `
          <tr>
            <td>${fila.numero}</td>
            <td>${formatearFecha(fila.fechaPago)}</td>
            <td>${formatearFecha(fila.fechaPagoPrincipal)}</td>
            <td>${fila.dias}</td>
            <td>${fila.diasAcumulados}</td>
            <td>${saldoInicial.toFixed(2)}</td>
            <td>${amortizacion.toFixed(2)}</td>
            <td>${interes.toFixed(2)}</td>
            <td>${segDesgravamen.toFixed(2)}</td>
            <td>${cuotaConstante.toFixed(2)}</td>
            <td>${saldoFinal.toFixed(2)}</td>
            <td>${fila.factor1.toFixed(2)}</td>
            <td>${fila.factor2.toFixed(2)}</td>
          </tr>
        `;

        // Se conserva el saldo final sin redondearlo.
        saldoInicial = saldoFinal;
      }

      // % Prima se conserva como proporción, sin redondear.
      const porcentajePrima = VPN / importe;

      // Seguro = % Prima * Importe.
      const seguro = porcentajePrima * importe;

      // Monto total a financiar:
      // (Importe + Seguro)
      // + (Importe + Seguro) * Seguro de desempleo.
      const baseFinanciamiento = importe + seguro;

      const montoTotalAFinanciar =
        baseFinanciamiento
        + baseFinanciamiento * tasaSeguroDesempleo;

      // Consola: valores internos sin limitar los decimales.
      console.log("VPN:", VPN);

      console.log(
        "% Prima:",
        (porcentajePrima * 100) + "%"
      );

      console.log("Seguro:", seguro);

      console.log(
        "Monto total a financiar:",
        montoTotalAFinanciar
      );

      document.getElementById("resultados").innerHTML = `
        <p>
          <strong>% Prima:</strong>
          ${(porcentajePrima * 100).toFixed(4)}%
        </p>

        <p>
          <strong>Seguro:</strong>
          ${seguro.toFixed(2)}
        </p>

        <p>
          <strong>Monto total a financiar:</strong>
          ${montoTotalAFinanciar.toFixed(2)}
        </p>

        <h3>Cálculo de prima</h3>

        <table>
          <thead>
            <tr>
              <th scope="col">N° de cuotas</th>
              <th scope="col">Fecha de pago</th>
              <th scope="col">Fecha de pago principal</th>
              <th scope="col">Días</th>
              <th scope="col">Días acumulados</th>
              <th scope="col">Saldo inicial</th>
              <th scope="col">Amortización</th>
              <th scope="col">Interés</th>
              <th scope="col">Seg de desgravamen</th>
              <th scope="col">Cuota constante</th>
              <th scope="col">Saldo final</th>
              <th scope="col">Factor1</th>
              <th scope="col">Factor 2</th>
            </tr>
          </thead>
          <tbody>
            ${filas}
          </tbody>
        </table>
      `;
    }

    function sumarUnMes(fecha) {
      const siguienteFecha = new Date(fecha);
      const diaAnterior = siguienteFecha.getDate();

      siguienteFecha.setDate(1);
      siguienteFecha.setMonth(
        siguienteFecha.getMonth() + 1
      );

      const ultimoDiaDelMes = new Date(
        siguienteFecha.getFullYear(),
        siguienteFecha.getMonth() + 1,
        0
      ).getDate();

      // Ejemplo: 30/01/2026 → 28/02/2026 → 28/03/2026.
      siguienteFecha.setDate(
        Math.min(diaAnterior, ultimoDiaDelMes)
      );

      return siguienteFecha;
    }

    function formatearFecha(fecha) {
      const dia = String(fecha.getDate()).padStart(2, "0");
      const mes = String(fecha.getMonth() + 1).padStart(2, "0");
      const anio = String(fecha.getFullYear()).padStart(4, "0");

      return `${dia}/${mes}/${anio}`;
    }

    function calcularDias(fechaInicial, fechaFinal) {
      const inicio = Date.UTC(
        fechaInicial.getFullYear(),
        fechaInicial.getMonth(),
        fechaInicial.getDate()
      );

      const fin = Date.UTC(
        fechaFinal.getFullYear(),
        fechaFinal.getMonth(),
        fechaFinal.getDate()
      );

      const milisegundosPorDia = 24 * 60 * 60 * 1000;

      return (fin - inicio) / milisegundosPorDia;
    }

    function limpiarResultados() {
      document.getElementById("resultados").innerHTML = "";
    }
  </script>
