    <script>
        const formulario = document.getElementById("formulario");
        const resultados = document.getElementById("resultados");
        const cronograma = document.getElementById("cronograma");

        // Suma meses conservando el día de la primera fecha de pago.
        // Si ese día no existe en el mes, usa el último día de ese mes.
        function sumarMeses(fechaInicial, meses) {
            const fecha = new Date(fechaInicial);
            const diaOriginal = fecha.getUTCDate();

            fecha.setUTCDate(1);
            fecha.setUTCMonth(fecha.getUTCMonth() + meses);

            const finDeMes = new Date(fecha);
            finDeMes.setUTCMonth(finDeMes.getUTCMonth() + 1, 0);

            fecha.setUTCDate(Math.min(diaOriginal, finDeMes.getUTCDate()));
            return fecha;
        }

        // Muestra la fecha en formato dd/mm/aaaa.
        function formatearFecha(fecha) {
            const dia = String(fecha.getUTCDate()).padStart(2, "0");
            const mes = String(fecha.getUTCMonth() + 1).padStart(2, "0");
            const anio = String(fecha.getUTCFullYear()).padStart(4, "0");

            return dia + "/" + mes + "/" + anio;
        }

        formulario.addEventListener("submit", function (evento) {
            evento.preventDefault();

            const plazo = Number(document.getElementById("plazo").value);
            const primerPago = document.getElementById("primer-pago").valueAsDate;
            const cuotaJulio = formulario.elements["cuota_julio"].checked;
            const cuotaDiciembre = formulario.elements["cuota_diciembre"].checked;

            // Si ambos checks están desmarcados, la cuota es Simple.
            // Con alguno marcado, el tipo queda vacío por ahora.
            const tipoCuota = !cuotaJulio && !cuotaDiciembre ? "Simple" : "";

            cronograma.textContent = "";

            for (let i = 0; i < plazo; i++) {
                const fechaPago = sumarMeses(primerPago, i);
                const fila = cronograma.insertRow();

                fila.insertCell().textContent = i + 1;
                fila.insertCell().textContent = formatearFecha(fechaPago);
                fila.insertCell().textContent = "0.00"; // Saldo inicial
                fila.insertCell().textContent = "0.00"; // Interés
                fila.insertCell().textContent = "0.00"; // Capital
                fila.insertCell().textContent = "0.00"; // Comisiones
                fila.insertCell().textContent = "0.00"; // Cuota
                fila.insertCell().textContent = "0.00"; // Saldo final
                fila.insertCell().textContent = tipoCuota;
            }
        });

        formulario.addEventListener("reset", function () {
            resultados.textContent = "";
            cronograma.textContent = "";
        });
    </script>
