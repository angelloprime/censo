    function cambiarProducto(producto) {
      for (const tipo of ["PP", "CD"]) {
        const activo = tipo === producto;

        document.getElementById(
          "panel-" + tipo.toLowerCase()
        ).hidden = !activo;

        const boton = document.getElementById(
          "boton-" + tipo.toLowerCase()
        );

        boton.classList.toggle("is-active", activo);
        boton.setAttribute("aria-pressed", String(activo));
      }
    }

    function convertirACentimosCD(texto) {
      const partes = String(texto).trim().match(
        /^\+?(\d+(?:\.\d*)?|\.\d+)(?:[eE]([+-]?\d+))?$/
      );

      if (!partes) {
        throw new Error("Ingresa un monto válido.");
      }

      const [entero, decimales = ""] =
        partes[1].split(".");

      const exponente = Number(partes[2] || "0");

      if (
        !Number.isInteger(exponente) ||
        Math.abs(exponente) > 1000
      ) {
        throw new Error(
          "El monto ingresado está fuera de rango."
        );
      }

      const digitos = BigInt(
        (entero || "0") + decimales
      );

      const escala =
        2 + exponente - decimales.length;

      if (escala >= 0) {
        return digitos * 10n ** BigInt(escala);
      }

      const divisor = 10n ** BigInt(-escala);

      if (digitos % divisor !== 0n) {
        throw new Error(
          "El monto ingresado debe tener como máximo dos decimales."
        );
      }

      return digitos / divisor;
    }

    function formatearCentimosCD(centimos) {
      const negativo = centimos < 0n;
      const valor = negativo ? -centimos : centimos;

      return (
        `${negativo ? "-" : ""}${valor / 100n}.` +
        String(valor % 100n).padStart(2, "0")
      );
    }

    function mostrarImporteCD(unidades) {
      return formatearCentimosCD(
        unidades / UNIDADES_POR_CENTIMO_CD
      );
    }

    function mostrarImporteExactoCD(unidades) {
      const negativo = unidades < 0n;
      const valor = negativo ? -unidades : unidades;

      const decimales = String(
        valor % ESCALA_IMPORTES_CD
      )
        .padStart(6, "0")
        .replace(/0+$/, "")
        .padEnd(2, "0");

      return (
        `${negativo ? "-" : ""}` +
        `${valor / ESCALA_IMPORTES_CD}.` +
        decimales
      );
    }

    function maximoComunDivisorCD(a, b) {
      while (b !== 0n) {
        [a, b] = [b, a % b];
      }

      return a;
    }

    // PROVISIÓN DE LOS PP DENTRO DE CD
    // =====================================================

    function calcularMontoPPCD(
      montoIngresado,
      porcentaje,
      conCash = false,
      totalidad = true
    ) {
      if (montoIngresado <= 0n) {
        throw new Error(
          "El monto debe ser mayor que cero."
        );
      }

      if (
        porcentaje !== "2" &&
        porcentaje !== "4"
      ) {
        throw new Error(
          "Selecciona un porcentaje de 2% o 4%."
        );
      }

      const tasa = BigInt(porcentaje);

      /*
        Totalidad desmarcada:
        el monto ingresado es el importe sin provisión.
        CASH no modifica esta fórmula.
      */
      if (!totalidad) {
        const montoBase =
          montoIngresado * UNIDADES_POR_CENTIMO_CD;

        const montoPorcentaje =
          montoBase * tasa / 100n;

        return {
          montoIngresado,
          porcentaje,
          conCash,
          totalidad,
          montoBase,
          montoPorcentaje,
          total: montoBase + montoPorcentaje,
          disponibleInicial: 0n,
          ajustePrecision: false
        };
      }

      /*
        Totalidad marcada:
        se conserva el cálculo anterior.
      */
      const pasoBase =
        100n / maximoComunDivisorCD(tasa, 100n);

      const pasoTotal =
        pasoBase * (100n + tasa) / 100n;

      const baseOriginal =
        (montoIngresado / pasoTotal) * pasoBase;

      const totalOriginal =
        baseOriginal +
        baseOriginal * tasa / 100n;

      const disponibleInicial =
        montoIngresado - totalOriginal;

      const baseCentimos =
        baseOriginal +
        (conCash ? disponibleInicial : 0n);

      let montoBase =
        baseCentimos * UNIDADES_POR_CENTIMO_CD;

      let montoPorcentaje =
        montoBase * tasa / 100n;

      let total =
        montoBase + montoPorcentaje;

      let ajustePrecision = false;

      /*
        Con CASH y Totalidad marcada,
        el total truncado a dos decimales
        debe coincidir con el monto ingresado.
      */
      if (
        conCash &&
        total / UNIDADES_POR_CENTIMO_CD !==
          montoIngresado
      ) {
        const baseMilesimas =
          (
            (montoIngresado + 1n) * 1000n - 1n
          ) / (100n + tasa);

        montoBase = baseMilesimas * 1000n;
        montoPorcentaje = montoBase * tasa / 100n;
        total = montoBase + montoPorcentaje;
        ajustePrecision = true;
      }

      return {
        montoIngresado,
        porcentaje,
        conCash,
        totalidad,
        montoBase,
        montoPorcentaje,
        total,
        disponibleInicial,
        ajustePrecision
      };
    }

    function leerMontosCD() {
      const filas = Array.from(
        document.querySelectorAll(
          "#cd-montos .monto-row"
        )
      );

      const montos = [];

      for (let i = 0; i < filas.length; i++) {
        const fila = filas[i];
        const campo = fila.querySelector(".cd-monto");

        if (!campo.reportValidity()) {
          return null;
        }

        try {
          const centimos =
            convertirACentimosCD(campo.value);

          if (centimos <= 0n) {
            throw new Error(
              "El monto debe ser mayor que cero."
            );
          }

          montos.push({
            tipo:
              fila.querySelector(".cd-tipo").value,

            porcentaje:
              fila.querySelector(".cd-porcentaje").value,

            totalidad:
              fila.querySelector(".cd-totalidad").checked,

            centimos
          });
        } catch (error) {
          alert(`Monto ${i + 1}: ${error.message}`);
          campo.focus();
          return null;
        }
      }

      return montos;
    }

    function calcularImportesCD(montos) {
      const conCash = montos.some(
        monto => monto.tipo === "CASH"
      );

      const detallesPP = [];

      let totalComprarPP = 0n;
      let totalTC = 0n;
      let totalCash = 0n;

      for (const monto of montos) {
        if (monto.tipo === "PP") {
          const detalle = calcularMontoPPCD(
            monto.centimos,
            monto.porcentaje,
            conCash,
            monto.totalidad
          );

          detallesPP.push(detalle);

          totalComprarPP +=
            detalle.total / UNIDADES_POR_CENTIMO_CD;

        } else if (monto.tipo === "TC") {
          totalTC += monto.centimos;

        } else if (monto.tipo === "CASH") {
          totalCash += monto.centimos;

        } else {
          throw new Error(
            "Se encontró un tipo de monto no válido."
          );
        }
      }

      const importeTotalCD =
        totalComprarPP + totalTC;

      const importePrestamo =
        importeTotalCD + totalCash;

      if (importePrestamo <= 0n) {
        throw new Error(
          "El importe para el préstamo debe ser mayor que cero."
        );
      }

      if (
        importePrestamo >
        BigInt(Number.MAX_SAFE_INTEGER)
      ) {
        throw new Error(
          "El importe total es demasiado grande para calcular el préstamo."
        );
      }

      return {
        conCash,
        detallesPP,
        totalComprarPP,
        totalTC,
        totalCash,
        importeTotalCD,
        importePrestamo
      };
    }

    function establecerFechaDesembolsoActual(prefijo = "") {

    function calcularDias(desde, hasta) {
      const a = Date.UTC(
        desde.getFullYear(),
        desde.getMonth(),
        desde.getDate()
      );

      const b = Date.UTC(
        hasta.getFullYear(),
        hasta.getMonth(),
        hasta.getDate()
      );

      return (b - a) / 86400000;
    }

    function validarFechas(prefijo = "") {
      const desembolso = document.getElementById(
        prefijo + "desembolso"
      ).value;

      const primerPago = document.getElementById(
        prefijo + "primer-pago"
      ).value;

      if (!desembolso || !primerPago) {
        return null;
      }

      const dias = calcularDias(
        new Date(desembolso + "T00:00:00"),
        new Date(primerPago + "T00:00:00")
      );

      if (!Number.isFinite(dias)) {
        alert(
          "Fechas inválidas.\n\n" +
          "No se pudieron interpretar las fechas ingresadas."
        );

        return null;
      }

      if (dias >= 30 && dias <= 60) {
        return dias;
      }

      let motivo;

      if (dias < 0) {
        motivo =
          "La fecha del primer pago es anterior al desembolso.";

      } else if (dias === 0) {
        motivo = "Las dos fechas son iguales.";

      } else if (dias < 30) {
        motivo =
          "El primer pago debe ser al menos 30 días después del desembolso.";

      } else {
        motivo =
          "El primer pago no puede superar los 60 días desde el desembolso.";
      }

      alert(
        `Fechas inválidas.\n\n${motivo}\n\n` +
        `Diferencia ingresada: ${dias} días.\n` +
        "Rango permitido: de 30 a 60 días."
      );

      return null;
    }

    function sumarUnMes(fecha) {
      const siguiente = new Date(fecha);
      const dia = siguiente.getDate();

      siguiente.setDate(1);
      siguiente.setMonth(
        siguiente.getMonth() + 1
      );

      const ultimoDia = new Date(
        siguiente.getFullYear(),
        siguiente.getMonth() + 1,
        0
      ).getDate();

      siguiente.setDate(
        Math.min(dia, ultimoDia)
      );

      return siguiente;
    }

    function crearCalendario(datos) {
      const filas = [];

      let fechaPago = new Date(
        datos.fechaPrimerPago + "T00:00:00"
      );

      let anterior = new Date(
        datos.fechaDesembolso + "T00:00:00"
      );

      let diasAcumulados = 0;

      for (
        let numero = 1;
        numero <= datos.cantidadCuotas;
        numero++
      ) {
        const fechaPagoPrincipal =
          new Date(fechaPago);

        if (fechaPagoPrincipal.getDay() === 0) {
          fechaPagoPrincipal.setDate(
            fechaPagoPrincipal.getDate() + 1
          );
        }

        const dias = calcularDias(
          anterior,
          fechaPagoPrincipal
        );

        diasAcumulados += dias;

        filas.push({
          numero,
          fechaPago: new Date(fechaPago),
          fechaPagoPrincipal,
          dias,
          diasAcumulados,
          saldoInicial: 0,
          amortizacion: 0,
          interes: 0,
          segDesgravamen: 0,
          portes: 0,
          cuotaConstante: 0,
          saldoFinal: 0,
          factor1: 0,
          factor2: 0
        });

        anterior = fechaPagoPrincipal;
        fechaPago = sumarUnMes(fechaPago);
      }

      return filas;
    }

    function esCuotaDoble(fecha, datos) {
      const mes = fecha.getMonth() + 1;

      return (
        (mes === 7 && datos.cuotaJulioActiva) ||
        (mes === 12 && datos.cuotaDiciembreActiva)
      );
    }

    function calcularPrima(datos) {
      const filas = crearCalendario(datos);

      let sumaFactor1 = 0;
      let sumaFactor2 = 0;

      for (const fila of filas) {
        const denominador = Math.pow(
          1 + datos.teaTotalPrima,
          fila.diasAcumulados / 360
        );

        fila.factor1 = (
          esCuotaDoble(
            fila.fechaPagoPrincipal,
            datos
          ) ? 2 : 1
        ) / denominador;

        fila.factor2 = 1 / denominador;

        sumaFactor1 += fila.factor1;
        sumaFactor2 += fila.factor2;
      }

      const cuotaPrima =
        datos.importe / sumaFactor1;

      let saldo = datos.importe;
      let VPN = 0;

      filas.forEach((fila, indice) => {
        fila.saldoInicial = saldo;

        fila.interes = (
          Math.pow(
            1 + datos.tea,
            fila.dias / 360
          ) - 1
        ) * saldo;

        fila.segDesgravamen = (
          Math.pow(
            1 + datos.tasaSeguroPrima,
            fila.dias / 30
          ) - 1
        ) * saldo;

        fila.cuotaConstante =
          cuotaPrima * (
            esCuotaDoble(
              fila.fechaPagoPrincipal,
              datos
            ) ? 2 : 1
          );

        if (indice === filas.length - 1) {
          fila.cuotaConstante =
            saldo +
            fila.interes +
            fila.segDesgravamen;
        }

        fila.amortizacion =
          fila.cuotaConstante -
          fila.segDesgravamen -
          fila.interes;

        fila.saldoFinal =
          saldo - fila.amortizacion;

        saldo = fila.saldoFinal;

        VPN += fila.segDesgravamen / Math.pow(
          1 + datos.tasaCostoCapital,
          fila.numero
        );
      });

      return {
        filas,
        sumaFactor1,
        sumaFactor2,
        cuotaPrima,
        VPN
      };
    }

    function calcularPrincipal(datos) {
      const filas = crearCalendario(datos);

      let sumaFactor1 = 0;
      let sumaFactor2 = 0;

      for (const fila of filas) {
        const denominador = Math.pow(
          1 + datos.tea,
          fila.diasAcumulados / 360
        );

        fila.factor1 = (
          esCuotaDoble(
            fila.fechaPago,
            datos
          ) ? 2 : 1
        ) / denominador;

        fila.factor2 = 1 / denominador;

        sumaFactor1 += fila.factor1;
        sumaFactor2 += fila.factor2;
      }

      const cuotaAPagar = (
        datos.montoTotalAFinanciar +
        datos.portes * sumaFactor2
      ) / sumaFactor1;

      let saldo =
        datos.montoTotalAFinanciar;

      filas.forEach((fila, indice) => {
        fila.saldoInicial = saldo;

        fila.interes = (
          Math.pow(
            1 + datos.tea,
            fila.dias / 360
          ) - 1
        ) * saldo;

        fila.portes = datos.portes;

        fila.cuotaConstante =
          cuotaAPagar * (
            esCuotaDoble(
              fila.fechaPago,
              datos
            ) ? 2 : 1
          );

        if (indice === filas.length - 1) {
          fila.cuotaConstante =
            saldo +
            fila.interes +
            fila.portes;
        }

        fila.amortizacion =
          fila.cuotaConstante -
          fila.portes -
          fila.interes;

        fila.saldoFinal =
          saldo - fila.amortizacion;

        saldo = fila.saldoFinal;
      });

      return {
        filas,
        sumaFactor1,
        sumaFactor2,
        cuotaAPagar
      };
    }

    function obtenerPorcentajePrima(tipo, VPN, importe) {
      if (tipo === "vida") {
        return 0.037;
      }

      if (
        tipo === "endosado" ||
        tipo === "sin-seguro"
      ) {
        return 0;
      }

      return VPN / importe;
    }

    function redondearACentimos(valor) {
      const [base, exponente = "0"] =
        Math.abs(valor).toString().split("e");

      const desplazado = Number(
        base + "e" + (Number(exponente) + 2)
      );

      return (
        Math.sign(valor) *
        Math.round(desplazado)
      );
    }

    function calcularVectorI(filasPrincipal, tcea) {
      const filas = filasPrincipal.map(fila => {
        const cuota =
          fila.amortizacion +
          fila.interes +
          fila.portes;

        const diasAcumulados =
          fila.diasAcumulados;

        const factor = Math.pow(
          1 + tcea,
          diasAcumulados / 360
        );

        const calculoVACentimos =
          redondearACentimos(cuota / factor);

        return {
          cuota,
          diasAcumulados,
          factor,
          calculoVACentimos,
          calculoVA: calculoVACentimos / 100
        };
      });

      const sumaVACentimos = filas.reduce(
        (suma, fila) =>
          suma + fila.calculoVACentimos,
        0
      );

      return {
        filas,
        sumaVACentimos,
        sumaVA: sumaVACentimos / 100
      };
    }

    function calcularTCEA(importe, filasPrincipal) {
      const objetivo =
        redondearACentimos(importe);

      function evaluar(tcea) {
        const vector = calcularVectorI(
          filasPrincipal,
          tcea
        );

        return {
          tcea,
          ...vector,
          diferenciaCentimos:
            vector.sumaVACentimos - objetivo
        };
      }

      let inferior = 0;
      let superior = 1;
      let mejor = evaluar(inferior);

      if (mejor.diferenciaCentimos <= 0) {
        return mejor;
      }

      let extremo = evaluar(superior);

      for (
        let i = 0;
        extremo.diferenciaCentimos > 0 && i < 100;
        i++
      ) {
        superior = superior * 2 + 1;
        extremo = evaluar(superior);
      }

      if (extremo.diferenciaCentimos > 0) {
        throw new Error(
          "No se pudo encontrar un intervalo para calcular la TCEA."
        );
      }

      if (
        Math.abs(extremo.diferenciaCentimos) <
        Math.abs(mejor.diferenciaCentimos)
      ) {
        mejor = extremo;
      }

      if (extremo.diferenciaCentimos === 0) {
        return extremo;
      }

      for (let i = 0; i < 200; i++) {
        const medio =
          (inferior + superior) / 2;

        if (
          medio === inferior ||
          medio === superior
        ) {
          break;
        }

        const resultado = evaluar(medio);

        if (
          Math.abs(resultado.diferenciaCentimos) <
          Math.abs(mejor.diferenciaCentimos)
        ) {
          mejor = resultado;
        }

        if (resultado.diferenciaCentimos === 0) {
          return resultado;
        }

        if (resultado.diferenciaCentimos > 0) {
          inferior = medio;
        } else {
          superior = medio;
        }
      }

      return mejor;
    }

    function calcularPrestamo(datos) {
      if (
        !Number.isFinite(datos.importe) ||
        datos.importe <= 0 ||
        !Number.isFinite(datos.tea) ||
        datos.tea < 0 ||
        !Number.isSafeInteger(datos.cantidadCuotas) ||
        datos.cantidadCuotas < 1
      ) {
        throw new Error(
          "Revisa el importe, la TEA y el plazo ingresados."
        );
      }

      const seguroPrima =
        PRIMAS_SEGURO[datos.tipoSeguro];

      const costoCapital = COSTO_CAPITAL;

      const tasaSeguroPrima =
        seguroPrima / 100;

      const tasaCostoCapital =
        costoCapital / 100;

      const teaTotalPrima = Math.pow(
        1 + (
          (
            Math.pow(
              1 + datos.tea,
              30 / 360
            ) - 1
          ) + tasaSeguroPrima
        ),
        12
      ) - 1;

      const datosCalculo = {
        ...datos,
        tasaSeguroPrima,
        tasaCostoCapital,
        teaTotalPrima
      };

      const resultadoPrima =
        calcularPrima(datosCalculo);

      const VPN = resultadoPrima.VPN;

      const porcentajePrima =
        obtenerPorcentajePrima(
          datos.tipoSeguro,
          VPN,
          datos.importe
        );

      const seguro =
        porcentajePrima * datos.importe;

      const baseFinanciamiento =
        datos.importe + seguro;

      const montoSeguroDesempleo =
        baseFinanciamiento *
        datos.tasaSeguroDesempleo;

      const montoTotalAFinanciar =
        baseFinanciamiento +
        montoSeguroDesempleo;

      const resultadoPrincipal =
        calcularPrincipal({
          ...datosCalculo,
          montoTotalAFinanciar
        });

      const cuotaAPagar =
        resultadoPrincipal.cuotaAPagar;

      if (
        ![
          porcentajePrima,
          seguro,
          montoSeguroDesempleo,
          montoTotalAFinanciar,
          cuotaAPagar
        ].every(Number.isFinite)
      ) {
        throw new Error(
          "Los datos superan el rango de cálculo. " +
          "Revisa la TEA, el importe y el plazo."
        );
      }

      const resultadoVector =
        calcularTCEA(
          datos.importe,
          resultadoPrincipal.filas
        );

      return {
        ...datosCalculo,
        seguroPrima,
        costoCapital,
        VPN,
        porcentajePrima,
        seguro,
        montoSeguroDesempleo,
        montoTotalAFinanciar,
        cuotaAPagar,
        TCEA: resultadoVector.tcea,
        resultadoPrima,
        resultadoPrincipal,
        resultadoVector
      };
    }

    function simular() {
      const campo =
        document.getElementById("importe");

      if (!campo.reportValidity()) {
        return;
      }

      const datos = leerDatosPrestamo();

      if (!datos) {
        return;
      }

      try {
        const resultado = calcularPrestamo({
          ...datos,
          importe: Number(campo.value)
        });

        mostrarSimulacion(resultado);
        registrarResultadosPrestamo(resultado, "PP");

      } catch (error) {
        alert(error.message);
      }
    }

    function simularCD() {
      limpiarResultadosCD();

      const datos =
        leerDatosPrestamo("cd-");

      if (!datos) {
        return;
      }

      const montos = leerMontosCD();

      if (!montos) {
        return;
      }

      try {
        const importes =
          calcularImportesCD(montos);

        const importe = Number(
          formatearCentimosCD(
            importes.importePrestamo
          )
        );

        const resultado = calcularPrestamo({
          ...datos,
          importe
        });

        mostrarSimulacion(
          resultado,
          "cd-",
          importes
        );

        registrarImportesCD(importes);
        registrarResultadosPrestamo(resultado, "CD");

      } catch (error) {
        alert(error.message);
      }
    }

    function formatearFecha(fecha) {
      return [
        String(fecha.getDate()).padStart(2, "0"),
        String(fecha.getMonth() + 1).padStart(2, "0"),
        String(fecha.getFullYear()).padStart(4, "0")
      ].join("/");
    }

    function mostrarDosDecimales(valor) {
      const texto = valor.toFixed(2);

      return texto === "-0.00"
        ? "0.00"
        : texto;
    }
