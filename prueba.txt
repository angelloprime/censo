  <script>
    "use strict";

    const $ = id => document.getElementById(id);

    // Formatea céntimos con dos decimales exactos.
    function money(cents) {
      return (cents / 100n).toString() + "." +
        (cents % 100n).toString().padStart(2, "0");
    }

    // Formatea unidades de 0.0001 sin redondear.
    // Conserva entre dos y cuatro decimales.
    function precise(units) {
      const whole = units / 10000n;

      let fraction = (units % 10000n)
        .toString()
        .padStart(4, "0");

      while (
        fraction.length > 2 &&
        fraction.endsWith("0")
      ) {
        fraction = fraction.slice(0, -1);
      }

      return whole.toString() + "." + fraction;
    }

    function parseAmount(value) {
      value = value.trim();

      if (!/^\d{1,15}([.,]\d{1,2})?$/.test(value)) {
        return null;
      }

      const [whole, fraction = ""] =
        value.replace(",", ".").split(".");

      return BigInt(whole) * 100n +
        BigInt(fraction.padEnd(2, "0"));
    }

    function clearResults() {
      ["base", "fee", "total", "remaining"].forEach(id => {
        $(id).textContent = "—";
      });

      $("summary").textContent = "";
      $("summary").hidden = true;
    }

    function update() {
      const raw = $("amount").value;
      const amount = parseAmount(raw);
      const rate = BigInt($("rate").value);
      const cash = $("cash").checked;

      $("percentage").textContent = rate + "%";
      $("error").textContent = "";
      $("error").hidden = true;
      $("remainingRow").hidden = cash;

      clearResults();

      if (amount === null) {
        if (raw.trim()) {
          $("error").textContent =
            "Ingresa un importe no negativo, sin separador de miles, " +
            "con hasta 15 dígitos enteros y 2 decimales.";

          $("error").hidden = false;
        }

        return;
      }

      if (cash) {
        calculateWithCash(amount, rate);
      } else {
        calculateWithoutCash(amount, rate);
      }
    }

    function calculateWithCash(amount, rate) {
      const divisor = 100n + rate;

      // Base en céntimos, tomando el siguiente céntimo
      // cuando la división no sea exacta.
      const base =
        (amount * 100n + divisor - 1n) / divisor;

      // Provisión y total en unidades de 0.0001.
      const feeUnits = base * rate;
      const totalUnits = base * 100n + feeUnits;

      // Compara enteros y primeros dos decimales,
      // descartando el resto sin redondear.
      const totalTruncatedToCents = totalUnits / 100n;

      if (totalTruncatedToCents !== amount) {
        $("error").textContent =
          "Este importe no permite una base de dos decimales " +
          "cuyo total con provisión coincida con el importe ingresado " +
          "hasta el segundo decimal. Para lograrlo, la base " +
          "necesitaría más de dos decimales.";

        $("error").hidden = false;
        return;
      }

      $("base").textContent = money(base);
      $("fee").textContent = precise(feeUnits);
      $("total").textContent = precise(totalUnits);

      $("summary").hidden = false;
      $("summary").innerHTML =
        `Monto sin provisión: <strong>${money(base)}</strong>, ` +
        `más <strong>${precise(feeUnits)}</strong> de provisión. ` +
        `Total: <strong>${precise(totalUnits)}</strong>. ` +
        `Tomando solo los primeros dos decimales, sin redondear: ` +
        `<strong>${money(totalTruncatedToCents)}</strong>.`;
    }

    function calculateWithoutCash(amount, rate) {
      // Bases válidas:
      // 2% → múltiplos de 0.50.
      // 4% → múltiplos de 0.25.
      const step = rate === 2n ? 50n : 25n;
      const totalStep = step + step * rate / 100n;

      // Mayor base válida sin superar el importe disponible.
      const base = (amount / totalStep) * step;
      const fee = base * rate / 100n;
      const total = base + fee;
      const remaining = amount - total;

      $("base").textContent = money(base);
      $("fee").textContent = money(fee);
      $("total").textContent = money(total);
      $("remaining").textContent = money(remaining);

      $("summary").hidden = false;

      if (base === 0n) {
        $("summary").textContent =
          "El importe disponible no alcanza para una venta positiva " +
          "con provisión exacta en céntimos.";

        return;
      }

      $("summary").innerHTML =
        `Puedes vender <strong>${money(base)}</strong> sin provisión. ` +
        `Al sumar <strong>${money(fee)}</strong> de provisión ` +
        `(${rate}%), el total es <strong>${money(total)}</strong>.`;
    }

    $("amount").addEventListener("input", update);
    $("rate").addEventListener("change", update);
    $("cash").addEventListener("change", update);

    update();
  </script>
