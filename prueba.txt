    <h3>Cronograma</h3>

    <table border="1">
      <thead>
        <tr>
          <th scope="col">N° de cuota</th>
          <th scope="col">Ciclo de facturación</th>
          <th scope="col">Vencim</th>
          <th scope="col">Días</th>
          <th scope="col">Días acum</th>
          <th scope="col">Factor</th>
          <th scope="col">Principal pendiente</th>
          <th scope="col">Amortización</th>
          <th scope="col">InterésY</th>
          <th scope="col">CuotaY</th>
          <th scope="col">Saldo final</th>
          <th scope="col">Amortización Principal</th>
          <th scope="col">Interés</th>
          <th scope="col">Cuota</th>
          <th scope="col">Seg de desgravamen</th>
          <th scope="col">Cuota + Seg desgravamen</th>
          <th scope="col">Último día de Pago (Referencial)</th>
        </tr>

  <script>
    const formulario = document.getElementById('formulario');
    const cuotas = document.getElementById('cuotas');
    const resultados = document.getElementById('resultados');

    const MILISEGUNDOS_POR_DIA = 24 * 60 * 60 * 1000;

    function formatearFecha(fecha) {
      const dia = String(fecha.getUTCDate()).padStart(2, '0');
      const mes = String(fecha.getUTCMonth() + 1).padStart(2, '0');
      const anio = String(fecha.getUTCFullYear()).padStart(4, '0');

      return `${dia}/${mes}/${anio}`;
    }

    function diferenciaDias(fechaFinal, fechaInicial) {
      return (
        (fechaFinal.getTime() - fechaInicial.getTime()) /
        MILISEGUNDOS_POR_DIA
      );
    }

    function redondear(valor, decimales) {
      const [base, exponente = '0'] =
        Math.abs(valor).toString().split('e');

      const desplazado = Number(
        base + 'e' + (Number(exponente) + decimales)
      );

      return (
        Math.sign(valor) *
        Math.round(desplazado) /
        (10 ** decimales)
      );
    }

    formulario.addEventListener('submit', function (evento) {
      evento.preventDefault();

      const importe = document.getElementById('importe').valueAsNumber;
      const plazo = document.getElementById('plazo').valueAsNumber;
      const ciclo = Number(document.getElementById('ciclo').value);
      const tea = document.getElementById('tea').valueAsNumber / 100;

      const fechaIngresada = document.getElementById('fecha').value;
      const fechaDesembolso = new Date(
        fechaIngresada + 'T00:00:00Z'
      );

      const mesesAdicionales =
        fechaDesembolso.getUTCDate() <= ciclo ? 0 : 1;

      const mesPrimeraCuota =
        fechaDesembolso.getUTCMonth() + mesesAdicionales;

      const cronograma = [];
      let vencimientoAnterior = null;
      let diasAcumulados = 0;

      cuotas.replaceChildren();

      // Primero se calculan las fechas, los días y los factores.
      for (let numero = 1; numero <= plazo; numero++) {

        const fechaFacturacion = new Date(fechaDesembolso);
        fechaFacturacion.setUTCDate(ciclo);
        fechaFacturacion.setUTCMonth(
          mesPrimeraCuota + numero - 1
        );

        // Vencim: restar 3 días y sumar 1 mes.
        const fechaVencim = new Date(fechaFacturacion);

        fechaVencim.setUTCDate(
          fechaVencim.getUTCDate() - 3
        );

        fechaVencim.setUTCMonth(
          fechaVencim.getUTCMonth() + 1
        );

        let dias;

        if (numero === 1) {
          dias = diferenciaDias(
            fechaVencim,
            fechaDesembolso
          ) + 1;
        } else {
          dias = diferenciaDias(
            fechaVencim,
            vencimientoAnterior
          );
        }

        diasAcumulados += dias;

        const factorSinRedondear =
          1 / Math.pow(1 + tea, diasAcumulados / 360);

        const factor =
          Math.round(factorSinRedondear * 1e9) / 1e9;

        cronograma.push({
          numero,
          fechaFacturacion,
          fechaVencim,
          dias,
          diasAcumulados,
          factor
        });

        vencimientoAnterior = fechaVencim;
      }

      // Sumar los factores sin redondear adicionalmente la suma.
      const sumaFactor = cronograma.reduce(function (suma, dato) {
        return suma + dato.factor;
      }, 0);

      console.log(
        'Suma de la columna Factor (sin redondeo adicional):',
        sumaFactor
      );

      // CuotaY redondeada a 2 decimales.
      const cuotaY = redondear(importe / sumaFactor, 2);

      let principalPendiente = importe;
      let amortizacionAcumulada = 0;
      let amortizacionPrincipalAcumulada = 0;

      cronograma.forEach(function (dato, indice) {
        const esUltimaFila = indice === cronograma.length - 1;

        let amortizacion;
        let interesY;

        if (esUltimaFila) {
          amortizacion = principalPendiente;
          interesY = redondear(cuotaY - amortizacion, 2);
        } else {
          interesY = principalPendiente * (
            Math.pow(1 + tea, dato.dias / 360) - 1
          );

          amortizacion = cuotaY - interesY;
        }

        const saldoFinal = principalPendiente - amortizacion;

        // Suma de Amortización desde la primera fila
        // hasta la fila actual, incluida.
        amortizacionAcumulada += amortizacion;

        // Amortización Principal:
        // si el acumulado de Amortización es <= 0, colocar 1.00;
        // en caso contrario, restar lo acumulado en las filas
        // anteriores de Amortización Principal.
        const amortizacionPrincipal =
          amortizacionAcumulada <= 0
            ? 1
            : amortizacionAcumulada - amortizacionPrincipalAcumulada;

        const vencimTexto = formatearFecha(dato.fechaVencim);

        const datosFila = [
          dato.numero,
          formatearFecha(dato.fechaFacturacion),
          vencimTexto,
          dato.dias,
          dato.diasAcumulados,
          dato.factor.toFixed(9),
          principalPendiente.toFixed(2),
          amortizacion.toFixed(2),
          interesY.toFixed(2),
          cuotaY.toFixed(2),
          saldoFinal.toFixed(2),
          amortizacionPrincipal.toFixed(2),
          '0.00', // Interés
          '0.00', // Cuota
          '0.00', // Seg de desgravamen
          '0.00', // Cuota + Seg desgravamen
          vencimTexto
        ];

        const fila = cuotas.insertRow();

        datosFila.forEach(function (valor) {
          fila.insertCell().textContent = valor;
        });

        // Acumulados que se utilizarán en la siguiente fila.
        principalPendiente = saldoFinal;
        amortizacionPrincipalAcumulada += amortizacionPrincipal;
      });
    });

    formulario.addEventListener('reset', function () {
      cuotas.replaceChildren();
      resultados.replaceChildren();
    });
  </script>
