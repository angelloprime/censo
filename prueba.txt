  <script>
    "use strict";

    const $ = id => document.getElementById(id);

    // Importes en céntimos, sin errores de punto flotante.
    function money(cents) {
      return (cents / 100n).toString() + "." +
        (cents % 100n).toString().padStart(2, "0");
    }

    function parseAmount(value) {
      value = value.trim();

      if (!/^\d{1,15}([.,]\d{1,2})?$/.test(value)) {
        return null;
      }

      const [whole, fraction = ""] =
        value.replace(",", ".").split(".");

      return BigInt(whole) * 100n +
        BigInt(fraction.padEnd(2, "0"));
    }

    function update() {
      const raw = $("amount").value;
      const amount = parseAmount(raw);
      const rate = BigInt($("rate").value);

      $("percentage").textContent = rate + "%";
      $("error").hidden = true;
      $("summary").hidden = true;

      if (amount === null) {
        ["base", "fee", "total", "remaining"].forEach(id => {
          $(id).textContent = "—";
        });

        if (raw.trim()) {
          $("error").textContent =
            "Ingresa un importe no negativo, sin separador de miles, " +
            "con hasta 15 dígitos enteros y 2 decimales.";
          $("error").hidden = false;
        }

        return;
      }

      // Al 2%, la base avanza en múltiplos de 0.50.
      // Al 4%, la base avanza en múltiplos de 0.25.
      // Así la provisión también es exacta en céntimos.
      const step = rate === 2n ? 50n : 25n;
      const totalStep = step + step * rate / 100n;

      // Mayor base válida cuyo total no supera el importe.
      const base = (amount / totalStep) * step;
      const fee = base * rate / 100n;
      const total = base + fee;
      const remaining = amount - total;

      $("base").textContent = money(base);
      $("fee").textContent = money(fee);
      $("total").textContent = money(total);
      $("remaining").textContent = money(remaining);

      $("summary").hidden = false;

      if (base === 0n) {
        $("summary").textContent =
          "El importe disponible no alcanza para una venta positiva " +
          "con provisión exacta en céntimos.";
        return;
      }

      $("summary").innerHTML =
        `Puedes vender <strong>${money(base)}</strong> sin provisión. ` +
        `Al sumar <strong>${money(fee)}</strong> de provisión ` +
        `(${rate}%), el total es <strong>${money(total)}</strong>.`;
    }

    $("amount").addEventListener("input", update);
    $("rate").addEventListener("change", update);

    update();
  </script>

    function establecerFechaDesembolsoActual() {
      // Se usa la fecha local del navegador.
      const hoy = new Date();

      const anio = String(hoy.getFullYear()).padStart(4, "0");
      const mes = String(hoy.getMonth() + 1).padStart(2, "0");
      const dia = String(hoy.getDate()).padStart(2, "0");

      const fechaActual = `${anio}-${mes}-${dia}`;

      const campoDesembolso =
        document.getElementById("desembolso");

      // defaultValue permite conservar esta fecha al resetear.
      campoDesembolso.defaultValue = fechaActual;
      campoDesembolso.value = fechaActual;
    }

    function validarFechas() {
      const campoDesembolso =
        document.getElementById("desembolso");

      const campoPrimerPago =
        document.getElementById("primer-pago");

      // Si falta una fecha, todavía no se compara.
      if (
        !campoDesembolso.value ||
        !campoPrimerPago.value
      ) {
        return null;
      }

      const fechaDesembolso = new Date(
        campoDesembolso.value + "T00:00:00"
      );

      const fechaPrimerPago = new Date(
        campoPrimerPago.value + "T00:00:00"
      );

      const cantidadDias = calcularDias(
        fechaDesembolso,
        fechaPrimerPago
      );

      if (!Number.isFinite(cantidadDias)) {
        alert(
          "Fechas inválidas.\n\n" +
          "No se pudieron interpretar las fechas ingresadas."
        );

        return null;
      }

      // 30 y 60 días sí son válidos.
      if (cantidadDias >= 30 && cantidadDias <= 60) {
        return cantidadDias;
      }

      let motivo;

      if (cantidadDias < 0) {
        motivo =
          "La fecha del primer pago es anterior " +
          "a la fecha de desembolso.";
      } else if (cantidadDias === 0) {
        motivo =
          "La fecha del primer pago coincide " +
          "con la fecha de desembolso.";
      } else if (cantidadDias < 30) {
        motivo =
          "La diferencia entre las fechas es menor " +
          "al mínimo permitido de 30 días.";
      } else {
        motivo =
          "La diferencia entre las fechas supera " +
          "el máximo permitido de 60 días.";
      }

      alert(
        "Fechas inválidas.\n\n" +
        motivo +
        "\n\nDiferencia ingresada: " +
        cantidadDias +
        " días." +
        "\nRango permitido: de 30 a 60 días."
      );

      return null;
    }

    function simular() {
      const campoTea = document.getElementById("tea");
      const campoImporte = document.getElementById("importe");
      const campoPlazo = document.getElementById("plazo");

      const campoDesembolso =
        document.getElementById("desembolso");

      const campoFecha =
        document.getElementById("primer-pago");

      const campos = [
        campoTea,
        campoImporte,
        campoPlazo,
        campoDesembolso,
        campoFecha
      ];

      for (const campo of campos) {
        if (!campo.reportValidity()) {
          return;
        }
      }

      // Se vuelve a validar antes de realizar cualquier cálculo.
      const cantidadDias = validarFechas();

      if (cantidadDias === null) {
        campoFecha.focus();
        return;
      }

      const primas = {
        "vida": 3.7000,
        "desgravamen": 0.1995,
        "desgravamen-devolucion": 0.2594,
        "sin-seguro": 0.0000,
        "endosado": 0.0000
      };

      const tipoSeguro =
        document.getElementById("seguro").value;

      const seguroPrima = primas[tipoSeguro];
      const costoCapital = 0.2871;

      const tea = Number(campoTea.value) / 100;
      const tasaSeguroPrima = seguroPrima / 100;
      const tasaCostoCapital = costoCapital / 100;

      const importe = Number(campoImporte.value);

      const tasaSeguroDesempleo =
        document.getElementById("seguro-desempleo").checked
          ? 0.04
          : 0;

      const portes =
        document.getElementById("estado-cuenta").checked
          ? 11
          : 0;

      const teaTotalPrima = Math.pow(
        1 + (
          (Math.pow(1 + tea, 30 / 360) - 1) +
          tasaSeguroPrima
        ),
        12
      ) - 1;

      const datos = {
        importe,
        tea,
        tasaSeguroPrima,
        tasaCostoCapital,
        teaTotalPrima,

        cantidadCuotas: Number(campoPlazo.value),

        fechaDesembolso: campoDesembolso.value,
        fechaPrimerPago: campoFecha.value,

        cuotaJulioActiva:
          document.getElementById("cuota-julio").checked,

        cuotaDiciembreActiva:
          document.getElementById("cuota-diciembre").checked
      };

      // Cálculo de prima: se conserva internamente.
      const resultadoPrima = calcularPrima(datos);

      const VPN = resultadoPrima.VPN;

      const porcentajePrima = obtenerPorcentajePrima(
        tipoSeguro,
        VPN,
        importe
      );

      const seguro = porcentajePrima * importe;

      const baseFinanciamiento = importe + seguro;

      const montoSeguroDesempleo =
        baseFinanciamiento * tasaSeguroDesempleo;

      const montoTotalAFinanciar =
        baseFinanciamiento + montoSeguroDesempleo;

      // Cálculo principal.
      const resultadoPrincipal = calcularPrincipal({
        ...datos,
        montoTotalAFinanciar,
        portes
      });

      const cuotaAPagar =
        resultadoPrincipal.cuotaAPagar;

      // Vector i: se conserva internamente para calcular la TCEA.
      let resultadoVector;

      try {
        resultadoVector = calcularTCEA(
          importe,
          resultadoPrincipal.filas
        );
      } catch (error) {
        alert(error.message);
        return;
      }

      const TCEA = resultadoVector.tcea;

      /* RESULTADOS EN CONSOLA */

      const primaFija =
        tipoSeguro === "vida" ||
        tipoSeguro === "endosado" ||
        tipoSeguro === "sin-seguro";

      console.log(
        "Seguro prima:",
        seguroPrima.toFixed(4) + "%"
      );

      console.log(
        "Costo capital:",
        costoCapital.toFixed(4) + "%"
      );

      console.log(
        "TEA total prima:",
        teaTotalPrima * 100 + "%"
      );

      console.log(
        "Suma Factor1 (Cálculo de prima):",
        resultadoPrima.sumaFactor1
      );

      console.log(
        "Suma Factor2 (Cálculo de prima):",
        resultadoPrima.sumaFactor2
      );

      console.log(
        "Cuota prima:",
        resultadoPrima.cuotaPrima
      );

      console.log("VPN:", VPN);

      console.log(
        "% Prima:",
        primaFija
          ? (porcentajePrima * 100).toFixed(4) + "%"
          : porcentajePrima * 100 + "%"
      );

      console.log("Seguro:", seguro);

      console.log(
        "Seguro de desempleo:",
        montoSeguroDesempleo
      );

      console.log(
        "Monto total a financiar:",
        montoTotalAFinanciar
      );

      console.log(
        "Suma Factor1 (Cronograma de pagos):",
        resultadoPrincipal.sumaFactor1
      );

      console.log(
        "Suma Factor2 (Cronograma de pagos):",
        resultadoPrincipal.sumaFactor2
      );

      console.log("Cuota a pagar:", cuotaAPagar);
      console.log("TCEA:", TCEA * 100 + "%");

      console.log(
        "Suma Cálculo VA:",
        resultadoVector.sumaVA
      );

      /* RESUMEN VISUAL */

      const observacionTCEA =
        resultadoVector.diferenciaCentimos === 0
          ? ""
          : `
            <div class="notice">
              <i class="bi bi-info-circle" aria-hidden="true"></i>

              <span>
                La suma de VA redondeados difiere del importe en
                ${(resultadoVector.diferenciaCentimos / 100).toFixed(2)}.
                La TCEA mostrada es una aproximación.
              </span>
            </div>
          `;

      const resultados =
        document.getElementById("resultados");

      resultados.innerHTML = `
        <div class="days-strip">
          <span>
            <i class="bi bi-calendar3" aria-hidden="true"></i>
            Cantidad de días
          </span>

          <strong>${cantidadDias}</strong>
        </div>

        <div class="summary-grid">
          <div class="result-item">
            <span class="result-label">% Prima</span>
            <strong class="result-value">
              ${(porcentajePrima * 100).toFixed(4)}%
            </strong>
          </div>

          <div class="result-item">
            <span class="result-label">Seguro</span>
            <strong class="result-value">
              ${seguro.toFixed(2)}
            </strong>
          </div>

          <div class="result-item">
            <span class="result-label">Seguro de desempleo</span>
            <strong class="result-value">
              ${montoSeguroDesempleo.toFixed(2)}
            </strong>
          </div>

          <div class="result-item">
            <span class="result-label">Monto total a financiar</span>
            <strong class="result-value">
              ${montoTotalAFinanciar.toFixed(2)}
            </strong>
          </div>

          <div class="result-item result-payment">
            <span class="result-label">Cuota a pagar</span>
            <strong class="result-value">
              ${cuotaAPagar.toFixed(2)}
            </strong>
          </div>

          <div class="result-item result-tcea">
            <span class="result-label">TCEA</span>
            <strong class="result-value">
              ${(TCEA * 100).toFixed(2)}%
            </strong>
          </div>
        </div>

        ${observacionTCEA}
      `;

      // Solo se genera la tabla visible del cronograma principal.
      document.getElementById("tablas").innerHTML =
        crearCronograma(resultadoPrincipal.filas);

      document.getElementById("estado-vacio").hidden = true;

      resultados.hidden = false;

      // Después de simular, el cronograma empieza oculto.
      establecerCronogramaVisible(false);

      document.getElementById("boton-cronograma").hidden = false;
    }

    function obtenerPorcentajePrima(
      tipoSeguro,
      VPN,
      importe
    ) {
      if (tipoSeguro === "vida") {
        return 3.7000 / 100;
      }

      if (
        tipoSeguro === "endosado" ||
        tipoSeguro === "sin-seguro"
      ) {
        return 0;
      }

      return VPN / importe;
    }

    /* =========================================================
       CÁLCULO PRINCIPAL
       ========================================================= */

    function calcularPrincipal(datos) {
      const filas = crearCalendario(datos);

      let sumaFactor1 = 0;
      let sumaFactor2 = 0;

      for (const fila of filas) {
        // Usa Fecha de pago original para julio y diciembre.
        const mes = fila.fechaPago.getMonth() + 1;

        fila.cuotaDoble =
          (mes === 7 && datos.cuotaJulioActiva) ||
          (mes === 12 && datos.cuotaDiciembreActiva);

        // Los factores siguen calculándose internamente.
        const denominador = Math.pow(
          1 + datos.tea,
          fila.diasAcumulados / 360
        );

        fila.factor1 =
          (fila.cuotaDoble ? 2 : 1) / denominador;

        fila.factor2 = 1 / denominador;

        sumaFactor1 += fila.factor1;
        sumaFactor2 += fila.factor2;
      }

      const cuotaAPagar = (
        datos.montoTotalAFinanciar +
        datos.portes * sumaFactor2
      ) / sumaFactor1;

      let saldoInicial = datos.montoTotalAFinanciar;

      for (const fila of filas) {
        fila.saldoInicial = saldoInicial;

        fila.interes = (
          Math.pow(
            1 + datos.tea,
            fila.dias / 360
          ) - 1
        ) * saldoInicial;

        fila.portes = datos.portes;

        fila.cuotaConstante =
          cuotaAPagar * (fila.cuotaDoble ? 2 : 1);

        if (fila.numero === datos.cantidadCuotas) {
          fila.cuotaConstante =
            saldoInicial + fila.interes + fila.portes;
        }

        fila.amortizacion =
          fila.cuotaConstante -
          fila.portes -
          fila.interes;

        fila.saldoFinal =
          saldoInicial - fila.amortizacion;

        saldoInicial = fila.saldoFinal;
      }

      return {
        filas,
        sumaFactor1,
        sumaFactor2,
        cuotaAPagar
      };
    }

    /* =========================================================
       CÁLCULO DE PRIMA — INTERNO
       ========================================================= */

    function calcularPrima(datos) {
      const filas = crearCalendario(datos);

      let sumaFactor1 = 0;
      let sumaFactor2 = 0;

      for (const fila of filas) {
        const mes =
          fila.fechaPagoPrincipal.getMonth() + 1;

        fila.cuotaDoble =
          (mes === 7 && datos.cuotaJulioActiva) ||
          (mes === 12 && datos.cuotaDiciembreActiva);

        const denominador = Math.pow(
          1 + datos.teaTotalPrima,
          fila.diasAcumulados / 360
        );

        fila.factor1 =
          (fila.cuotaDoble ? 2 : 1) / denominador;

        fila.factor2 = 1 / denominador;

        sumaFactor1 += fila.factor1;
        sumaFactor2 += fila.factor2;
      }

      const cuotaPrima =
        datos.importe / sumaFactor1;

      let saldoInicial = datos.importe;
      let VPN = 0;

      for (const fila of filas) {
        fila.saldoInicial = saldoInicial;

        fila.interes = (
          Math.pow(
            1 + datos.tea,
            fila.dias / 360
          ) - 1
        ) * saldoInicial;

        fila.segDesgravamen = (
          Math.pow(
            1 + datos.tasaSeguroPrima,
            fila.dias / 30
          ) - 1
        ) * saldoInicial;

        fila.cuotaConstante =
          cuotaPrima * (fila.cuotaDoble ? 2 : 1);

        if (fila.numero === datos.cantidadCuotas) {
          fila.cuotaConstante =
            saldoInicial +
            fila.interes +
            fila.segDesgravamen;
        }

        fila.amortizacion =
          fila.cuotaConstante -
          fila.segDesgravamen -
          fila.interes;

        fila.saldoFinal =
          saldoInicial - fila.amortizacion;

        VPN += fila.segDesgravamen / Math.pow(
          1 + datos.tasaCostoCapital,
          fila.numero
        );

        saldoInicial = fila.saldoFinal;
      }

      return {
        filas,
        sumaFactor1,
        sumaFactor2,
        cuotaPrima,
        VPN
      };
    }

    /* =========================================================
       VECTOR I Y TCEA — INTERNOS
       ========================================================= */

    function redondearACentimos(valor) {
      const [base, exponente = "0"] =
        Math.abs(valor).toString().split("e");

      const desplazado = Number(
        base + "e" + (Number(exponente) + 2)
      );

      return Math.sign(valor) * Math.round(desplazado);
    }

    function calcularVectorI(filasPrincipal, tcea) {
      const filas = filasPrincipal.map(fila => {
        const cuota =
          fila.amortizacion +
          fila.interes +
          fila.portes;

        const diasAcumulados =
          fila.diasAcumulados;

        const factor = Math.pow(
          1 + tcea,
          diasAcumulados / 360
        );

        // Cálculo VA sí se redondea a dos decimales internamente.
        const calculoVACentimos =
          redondearACentimos(cuota / factor);

        const calculoVA = calculoVACentimos / 100;

        return {
          cuota,
          diasAcumulados,
          factor,
          calculoVA,
          calculoVACentimos
        };
      });

      const sumaVACentimos = filas.reduce(
        (suma, fila) => suma + fila.calculoVACentimos,
        0
      );

      return {
        filas,
        sumaVACentimos,
        sumaVA: sumaVACentimos / 100
      };
    }

    function calcularTCEA(importe, filasPrincipal) {
      const objetivoCentimos =
        redondearACentimos(importe);

      function evaluar(tcea) {
        const vector =
          calcularVectorI(filasPrincipal, tcea);

        return {
          tcea,
          ...vector,
          diferenciaCentimos:
            vector.sumaVACentimos - objetivoCentimos
        };
      }

      let inferior = 0;
      let superior = 1;
      let mejor = evaluar(inferior);

      if (mejor.diferenciaCentimos <= 0) {
        return mejor;
      }

      let extremoSuperior = evaluar(superior);

      for (
        let intento = 0;
        extremoSuperior.diferenciaCentimos > 0 &&
        intento < 100;
        intento++
      ) {
        superior = superior * 2 + 1;
        extremoSuperior = evaluar(superior);
      }

      if (extremoSuperior.diferenciaCentimos > 0) {
        throw new Error(
          "No se pudo encontrar un intervalo para calcular la TCEA."
        );
      }

      if (
        Math.abs(extremoSuperior.diferenciaCentimos) <
        Math.abs(mejor.diferenciaCentimos)
      ) {
        mejor = extremoSuperior;
      }

      if (extremoSuperior.diferenciaCentimos === 0) {
        return extremoSuperior;
      }

      for (let intento = 0; intento < 200; intento++) {
        const medio = (inferior + superior) / 2;

        if (medio === inferior || medio === superior) {
          break;
        }

        const resultado = evaluar(medio);

        if (
          Math.abs(resultado.diferenciaCentimos) <
          Math.abs(mejor.diferenciaCentimos)
        ) {
          mejor = resultado;
        }

        if (resultado.diferenciaCentimos === 0) {
          return resultado;
        }

        if (resultado.diferenciaCentimos > 0) {
          inferior = medio;
        } else {
          superior = medio;
        }
      }

      return mejor;
    }

    /* =========================================================
       CALENDARIO Y FECHAS
       ========================================================= */

    function crearCalendario(datos) {
      const filas = [];

      let fechaPago = new Date(
        datos.fechaPrimerPago + "T00:00:00"
      );

      let fechaAnterior = new Date(
        datos.fechaDesembolso + "T00:00:00"
      );

      let diasAcumulados = 0;

      for (
        let numero = 1;
        numero <= datos.cantidadCuotas;
        numero++
      ) {
        const fechaPagoPrincipal =
          new Date(fechaPago);

        if (fechaPagoPrincipal.getDay() === 0) {
          fechaPagoPrincipal.setDate(
            fechaPagoPrincipal.getDate() + 1
          );
        }

        const dias = calcularDias(
          fechaAnterior,
          fechaPagoPrincipal
        );

        diasAcumulados += dias;

        filas.push({
          numero,
          fechaPago: new Date(fechaPago),
          fechaPagoPrincipal,
          dias,
          diasAcumulados
        });

        fechaAnterior = fechaPagoPrincipal;
        fechaPago = sumarUnMes(fechaPago);
      }

      return filas;
    }

    function sumarUnMes(fecha) {
      const siguienteFecha = new Date(fecha);
      const diaAnterior = siguienteFecha.getDate();

      siguienteFecha.setDate(1);

      siguienteFecha.setMonth(
        siguienteFecha.getMonth() + 1
      );

      const ultimoDiaDelMes = new Date(
        siguienteFecha.getFullYear(),
        siguienteFecha.getMonth() + 1,
        0
      ).getDate();

      siguienteFecha.setDate(
        Math.min(diaAnterior, ultimoDiaDelMes)
      );

      return siguienteFecha;
    }

    function calcularDias(fechaInicial, fechaFinal) {
      const inicio = Date.UTC(
        fechaInicial.getFullYear(),
        fechaInicial.getMonth(),
        fechaInicial.getDate()
      );

      const fin = Date.UTC(
        fechaFinal.getFullYear(),
        fechaFinal.getMonth(),
        fechaFinal.getDate()
      );

      return (fin - inicio) / (24 * 60 * 60 * 1000);
    }

    function formatearFecha(fecha) {
      const dia = String(
        fecha.getDate()
      ).padStart(2, "0");

      const mes = String(
        fecha.getMonth() + 1
      ).padStart(2, "0");

      const anio = String(
        fecha.getFullYear()
      ).padStart(4, "0");

      return `${dia}/${mes}/${anio}`;
    }
