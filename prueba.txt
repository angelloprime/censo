  <script>
    "use strict";

    // Las tasas se expresan en forma decimal:
    // 0.001995 equivale a 0.1995%.
    const PARAMETROS = {
      costoCapitalMensual: 0.002871,
      comisionFisica: 11,

      // Cada fila:
      // [desde días, hasta días, tarifa titular, tarifa con cónyuge].
      vida: {
        pp: [
          [0, 365, 0.00005930, 0.00011864],
          [366, 1095, 0.00003410, 0.00006825],
          [1096, 1825, 0.00003190, 0.00006387],
          [1826, 2190, 0.00002200, 0.00004407],
          [2191, 3650, 0.00001750, 0.00003504]
        ],
        convenio: [
          [0, 365, 0.00016659, 0.00033317],
          [366, 1095, 0.00006244, 0.00012488],
          [1096, 1825, 0.00003032, 0.00006064],
          [1826, 2190, 0.00002214, 0.00004429],
          [2191, 3650, 0.00001888, 0.00003777]
        ]
      },

      // Cada lista:
      // [tarifa titular, tarifa con cónyuge].
      desgravamen: {
        pp: {
          "Desgravamen": [0.001995, 0.003990],
          "Desgravamen con Devolución": [0.002594, 0.005188]
        },
        convenio: {
          "Desgravamen": [0.001832, 0.003664],
          "Desgravamen con Devolución": [0.002342, 0.004684]
        }
      }
    };

    const DIA = 86400000;

    const formatoNumero = new Intl.NumberFormat("es-PE", {
      minimumFractionDigits: 2,
      maximumFractionDigits: 2
    });

    const sumar = valores =>
      valores.reduce((total, valor) => total + valor, 0);

    const numero = valor =>
      formatoNumero.format(Math.abs(valor) < 0.005 ? 0 : valor);

    const moneda = valor => "S/ " + numero(valor);

    const diasEntre = (final, inicial) =>
      Math.round((final - inicial) / DIA);

    // UTC evita que la zona horaria cambie las fechas
    // o la cantidad de días entre ellas.
    function leerFecha(texto) {
      return new Date(texto + "T00:00:00Z");
    }

    function mostrarFecha(fecha) {
      const dia = String(fecha.getUTCDate()).padStart(2, "0");
      const mes = String(fecha.getUTCMonth() + 1).padStart(2, "0");

      return dia + "/" + mes + "/" + fecha.getUTCFullYear();
    }

    // Suma meses a la fecha base y limita el día
    // al último disponible del mes.
    function sumarMeses(fecha, meses) {
      const primero = new Date(
        Date.UTC(
          fecha.getUTCFullYear(),
          fecha.getUTCMonth() + meses,
          1
        )
      );

      const ultimoDia = new Date(
        Date.UTC(
          primero.getUTCFullYear(),
          primero.getUTCMonth() + 1,
          0
        )
      ).getUTCDate();

      return new Date(
        Date.UTC(
          primero.getUTCFullYear(),
          primero.getUTCMonth(),
          Math.min(fecha.getUTCDate(), ultimoDia)
        )
      );
    }

    // Si el vencimiento cae domingo, utiliza el lunes siguiente.
    function ajustarDomingo(fecha) {
      return new Date(
        fecha.getTime() + (fecha.getUTCDay() === 0 ? DIA : 0)
      );
    }

    // Redondea a dos decimales el importe del seguro de desempleo.
    function redondearDos(valor) {
      const [mantisa, exponente = "0"] = String(valor).split("e");

      const entero = Math.round(
        Number(mantisa + "e" + (Number(exponente) + 2))
      );

      const [mantisaFinal, exponenteFinal = "0"] =
        String(entero).split("e");

      return Number(
        mantisaFinal + "e" + (Number(exponenteFinal) - 2)
      );
    }

    // Busca la tasa anual r que cumple:
    // monto = suma(pago_i / (1 + r)^(días acumulados_i / 365)).
    // Luego la convierte a base 360: (1 + r)^(360/365) - 1.
    function calcularTcea(monto, pagos, diasAcumulados) {
      const valorPresente = tasa =>
        sumar(
          pagos.map((pago, i) =>
            pago / Math.pow(
              1 + tasa,
              diasAcumulados[i] / 365
            )
          )
        ) - monto;

      let inferior = -0.999999999;
      let superior = 1;

      while (
        valorPresente(superior) > 0 &&
        superior < 1e16
      ) {
        superior = superior * 2 + 1;
      }

      if (
        valorPresente(inferior) < 0 ||
        valorPresente(superior) > 0
      ) {
        return 0;
      }

      for (let i = 0; i < 120; i++) {
        const medio = (inferior + superior) / 2;

        if (valorPresente(medio) > 0) {
          inferior = medio;
        } else {
          superior = medio;
        }
      }

      return Math.pow(
        1 + (inferior + superior) / 2,
        360 / 365
      ) - 1;
    }

    function simular(datos) {
      const {
        producto,
        monto,
        plazo,
        tea,
        desembolso,
        primerPago,
        seguro,
        cobertura,
        estadoCuenta,
        julio,
        diciembre,
        desempleo
      } = datos;

      // Genera las fechas mensuales y traslada los domingos
      // al lunes para el cálculo.
      const fechas = Array.from(
        { length: plazo },
        (_, i) => sumarMeses(primerPago, i)
      );

      const fechasAjustadas = fechas.map(ajustarDomingo);

      // Primer período:
      // días = primer vencimiento ajustado - desembolso.
      // Siguientes períodos:
      // días = vencimiento actual - vencimiento anterior.
      const dias = fechasAjustadas.map((fecha, i) =>
        diasEntre(
          fecha,
          i === 0 ? desembolso : fechasAjustadas[i - 1]
        )
      );

      const acumulados = fechasAjustadas.map(fecha =>
        diasEntre(fecha, desembolso)
      );

      const factores = fechas.map(fecha => {
        const mes = fecha.getUTCMonth() + 1;

        return mes === 7
          ? julio
          : mes === 12
            ? diciembre
            : 1;
      });

      const esDesgravamen =
        seguro === "Desgravamen" ||
        seguro === "Desgravamen con Devolución";

      let primaPrincipal = 0;

      if (seguro === "Vida") {
        const duracion = acumulados[plazo - 1];
        const indice = cobertura === "titular" ? 2 : 3;

        const tarifa = sumar(
          PARAMETROS.vida[producto]
            .filter(fila =>
              duracion >= fila[0] &&
              duracion <= fila[1]
            )
            .map(fila => fila[indice])
        );

        // Prima de vida =
        // monto × tarifa diaria × duración total en días.
        primaPrincipal = monto * (tarifa * duracion);

      } else if (esDesgravamen) {
        const tarifa =
          PARAMETROS.desgravamen[producto][seguro][
            cobertura === "titular" ? 0 : 1
          ];

        const tem = Math.pow(1 + tea, 1 / 12) - 1;

        // Cuota auxiliar =
        // monto / suma(
        //   factor / (1 + TEM + tarifa)^(días acumulados / 30)
        // ).
        const cuotaAuxiliar = monto / sumar(
          factores.map((factor, i) =>
            factor / Math.pow(
              1 + tem + tarifa,
              acumulados[i] / 30
            )
          )
        );

        let saldoAuxiliar = monto;
        const primasMensuales = [];

        for (let i = 0; i < plazo; i++) {
          // Interés del período =
          // saldo × ((1 + TEA)^(días / 360) - 1).
          const interesAuxiliar =
            Math.pow(1 + tea, dias[i] / 360) - 1;

          // Prima del período =
          // saldo × ((1 + tarifa)^(días / 30) - 1).
          const tasaPeriodoSeguro =
            Math.pow(1 + tarifa, dias[i] / 30) - 1;

          primasMensuales.push(
            tasaPeriodoSeguro * saldoAuxiliar
          );

          saldoAuxiliar = Math.max(
            0,
            saldoAuxiliar *
              (1 + interesAuxiliar + tasaPeriodoSeguro) -
              cuotaAuxiliar * factores[i]
          );
        }

        // Valor presente de primas =
        // suma(prima_i / (1 + costo mensual)^i).
        const valorPresentePrimas = sumar(
          primasMensuales.map((prima, i) =>
            prima / Math.pow(
              1 + PARAMETROS.costoCapitalMensual,
              i + 1
            )
          )
        );

        primaPrincipal =
          monto * (valorPresentePrimas / monto);
      }

      // Desempleo =
      // (monto + prima principal) × tasa de desempleo.
      const adicionalDesempleo = redondearDos(
        (monto + primaPrincipal) * desempleo
      );

      // Monto financiado =
      // monto + prima principal + desempleo.
      const montoFinanciado =
        monto + primaPrincipal + adicionalDesempleo;

      const comision =
        estadoCuenta === "fisico"
          ? PARAMETROS.comisionFisica
          : 0;

      // Cuota base =
      // monto financiado / suma(
      //   factor / (1 + TEA)^(días acumulados / 360)
      // ).
      const cuotaBase = montoFinanciado / sumar(
        factores.map((factor, i) =>
          factor / Math.pow(
            1 + tea,
            acumulados[i] / 360
          )
        )
      );

      let saldo = montoFinanciado;

      const cronograma = [];
      const flujosTcea = [];

      for (let i = 0; i < plazo; i++) {
        const interes =
          saldo * (Math.pow(1 + tea, dias[i] / 360) - 1);

        const pagoSinComision = Math.min(
          saldo + interes,
          cuotaBase * factores[i]
        );

        const capital = Math.max(
          0,
          pagoSinComision - interes
        );

        const saldoFinal = Math.max(
          0,
          saldo - capital
        );

        cronograma.push({
          numero: i + 1,
          fecha: fechas[i],
          saldoInicial: saldo,
          interes,
          capital,
          comisiones: comision,
          cuota: pagoSinComision + comision,
          saldoFinal,
          tipo: factores[i] === 2 ? "Doble" : "Simple"
        });

        // Flujos de pago utilizados para hallar la TCEA.
        flujosTcea.push(
          cuotaBase * factores[i] + comision
        );

        saldo = saldoFinal;
      }

      return {
        cuotaEstimada: cuotaBase + comision,
        montoSolicitado: monto,
        primasSeguros: primaPrincipal + adicionalDesempleo,
        montoFinanciado,

        intereses: sumar(
          cronograma.map(fila => fila.interes)
        ),

        comisiones: sumar(
          cronograma.map(fila => fila.comisiones)
        ),

        totalPagar: sumar(
          cronograma.map(fila => fila.cuota)
        ),

        tcea: calcularTcea(
          monto,
          flujosTcea,
          acumulados
        ),

        adicionalDesempleo,
        cronograma
      };
    }

    const formulario = document.getElementById("formulario");
    const mensaje = document.getElementById("mensaje");
    const cuerpoCronograma = document.getElementById("cronograma");
    const tablaCronograma = document.getElementById("tablaCronograma");

    const valor = id =>
      document.getElementById(id).value;

    const marcado = id =>
      document.getElementById(id).checked;

    function leerDatos() {
      return {
        // Producto fijo: Prestabono / Libre Disponibilidad.
        // Cobertura fija: Titular.
        producto: "pp",
        cobertura: "titular",

        monto: Number(valor("monto")),
        plazo: Number(valor("plazo")),
        tea: Number(valor("tea")) / 100,
        desembolso: leerFecha(valor("desembolso")),
        primerPago: leerFecha(valor("primerPago")),
        seguro: valor("seguro"),

        estadoCuenta: marcado("estadoCuenta")
          ? "fisico"
          : "digital",

        julio: marcado("julio") ? 2 : 1,
        diciembre: marcado("diciembre") ? 2 : 1,
        desempleo: marcado("desempleo") ? 0.04 : 0
      };
    }

    function limpiarResultados() {
      document.querySelectorAll("output").forEach(salida => {
        salida.textContent = "—";
      });

      cuerpoCronograma.replaceChildren();
      tablaCronograma.hidden = true;
    }

    // Al modificar cualquier dato, se descarta la simulación anterior.
    function invalidarResultados() {
      mensaje.textContent = "";
      limpiarResultados();
    }

    function calcular() {
      invalidarResultados();

      if (!formulario.checkValidity()) {
        return;
      }

      const datos = leerDatos();

      if (datos.primerPago <= datos.desembolso) {
        limpiarResultados();

        mensaje.textContent =
          "El primer pago debe ser posterior a la fecha de desembolso.";

        return;
      }

      const resultado = simular(datos);

      const tieneValoresInvalidos =
        Object.entries(resultado).some(([clave, dato]) =>
          clave !== "cronograma" &&
          !Number.isFinite(dato)
        );

      if (tieneValoresInvalidos) {
        limpiarResultados();

        mensaje.textContent =
          "No se pudo calcular con los valores ingresados. Revisa los datos.";

        return;
      }

      Object.entries(resultado).forEach(([id, dato]) => {
        if (id !== "cronograma") {
          document.getElementById(id).textContent =
            id === "tcea"
              ? numero(dato * 100) + "%"
              : moneda(dato);
        }
      });

      const fragmento = document.createDocumentFragment();

      resultado.cronograma.forEach(fila => {
        const tr = document.createElement("tr");

        const celdas = [
          fila.numero,
          mostrarFecha(fila.fecha),
          numero(fila.saldoInicial),
          numero(fila.interes),
          numero(fila.capital),
          numero(fila.comisiones),
          numero(fila.cuota),
          numero(fila.saldoFinal),
          fila.tipo
        ];

        celdas.forEach(texto => {
          const td = document.createElement("td");
          td.textContent = texto;
          tr.appendChild(td);
        });

        fragmento.appendChild(tr);
      });

      cuerpoCronograma.replaceChildren(fragmento);
      tablaCronograma.hidden = false;
    }

    // La simulación se ejecuta únicamente al enviar el formulario.
    formulario.addEventListener("submit", evento => {
      evento.preventDefault();
      calcular();
    });

    // Editar o limpiar solo borra los resultados; no recalcula.
    formulario.addEventListener("input", invalidarResultados);
    formulario.addEventListener("change", invalidarResultados);
    formulario.addEventListener("reset", invalidarResultados);

    // Al abrir la página, los resultados y el cronograma están vacíos.
    invalidarResultados();
  </script>
