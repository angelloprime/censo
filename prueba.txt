    function simular() {
      const campoTea = document.getElementById("tea");
      const campoImporte = document.getElementById("importe");
      const campoPlazo = document.getElementById("plazo");
      const campoDesembolso = document.getElementById("desembolso");
      const campoFecha = document.getElementById("primer-pago");

      const campos = [
        campoTea,
        campoImporte,
        campoPlazo,
        campoDesembolso,
        campoFecha
      ];

      for (const campo of campos) {
        if (!campo.reportValidity()) {
          return;
        }
      }

      if (campoFecha.value <= campoDesembolso.value) {
        campoFecha.setCustomValidity(
          "La fecha del primer pago debe ser posterior al desembolso."
        );
        campoFecha.reportValidity();
        campoFecha.setCustomValidity("");
        return;
      }

      const primas = {
        "vida": 3.7000,
        "desgravamen": 0.1995,
        "desgravamen-devolucion": 0.2594,
        "sin-seguro": 0.0000,
        "endosado": 0.0000
      };

      const tipoSeguro = document.getElementById("seguro").value;
      const seguroPrima = primas[tipoSeguro];
      const costoCapital = 0.2871;

      const tea = Number(campoTea.value) / 100;
      const tasaSeguroPrima = seguroPrima / 100;
      const tasaCostoCapital = costoCapital / 100;
      const importe = Number(campoImporte.value);

      const tasaSeguroDesempleo =
        document.getElementById("seguro-desempleo").checked
          ? 0.04
          : 0;

      const portes =
        document.getElementById("estado-cuenta").checked
          ? 11
          : 0;

      const teaTotalPrima = Math.pow(
        1 + (
          (Math.pow(1 + tea, 30 / 360) - 1) +
          tasaSeguroPrima
        ),
        12
      ) - 1;

      const datos = {
        importe,
        tea,
        tasaSeguroPrima,
        tasaCostoCapital,
        teaTotalPrima,
        cantidadCuotas: Number(campoPlazo.value),
        fechaDesembolso: campoDesembolso.value,
        fechaPrimerPago: campoFecha.value,
        cuotaJulioActiva:
          document.getElementById("cuota-julio").checked,
        cuotaDiciembreActiva:
          document.getElementById("cuota-diciembre").checked
      };

      const resultadoPrima = calcularPrima(datos);

      const VPN = resultadoPrima.VPN;

      // Selección del porcentaje que se utilizará para el seguro.
      const porcentajePrima = obtenerPorcentajePrima(
        tipoSeguro,
        VPN,
        importe
      );

      const seguro = porcentajePrima * importe;

      const baseFinanciamiento = importe + seguro;

      const montoSeguroDesempleo =
        baseFinanciamiento * tasaSeguroDesempleo;

      const montoTotalAFinanciar =
        baseFinanciamiento + montoSeguroDesempleo;

      const resultadoPrincipal = calcularPrincipal({
        ...datos,
        montoTotalAFinanciar,
        portes
      });

      const cuotaAPagar = resultadoPrincipal.cuotaAPagar;

      let resultadoVector;

      try {
        resultadoVector = calcularTCEA(
          importe,
          resultadoPrincipal.filas
        );
      } catch (error) {
        alert(error.message);
        return;
      }

      const TCEA = resultadoVector.tcea;

      console.log(
        "Seguro prima:",
        seguroPrima.toFixed(4) + "%"
      );

      console.log(
        "Costo capital:",
        costoCapital.toFixed(4) + "%"
      );

      console.log(
        "Suma Factor1 (Cálculo de prima):",
        resultadoPrima.sumaFactor1

    function obtenerPorcentajePrima(tipoSeguro, VPN, importe) {
      if (tipoSeguro === "vida") {
        return 3.7000 / 100;
      }

      if (
        tipoSeguro === "endosado" ||
        tipoSeguro === "sin-seguro"
      ) {
        return 0;
      }

      // Seguro de desgravamen y desgravamen con devolución.
      return VPN / importe;
    }

    function calcularPrincipal(datos) {
      const filas = crearCalendario(datos);

      let sumaFactor1 = 0;
      let sumaFactor2 = 0;

      for (const fila of filas) {
        // Se utiliza Fecha de pago para determinar el mes.
        const mes = fila.fechaPago.getMonth() + 1;

        fila.cuotaDoble =
          (mes === 7 && datos.cuotaJulioActiva) ||
          (mes === 12 && datos.cuotaDiciembreActiva);

        const denominador = Math.pow(
          1 + datos.tea,
          fila.diasAcumulados / 360
        );

        fila.factor1 = (fila.cuotaDoble ? 2 : 1) / denominador;
        fila.factor2 = 1 / denominador;

        sumaFactor1 += fila.factor1;
        sumaFactor2 += fila.factor2;
      }

      const cuotaAPagar = (
        datos.montoTotalAFinanciar +
        datos.portes * sumaFactor2
      ) / sumaFactor1;

      let saldoInicial = datos.montoTotalAFinanciar;

      for (const fila of filas) {
        fila.saldoInicial = saldoInicial;

        fila.interes = (
          Math.pow(1 + datos.tea, fila.dias / 360) - 1
        ) * saldoInicial;

        fila.portes = datos.portes;

        fila.cuotaConstante =
          cuotaAPagar * (fila.cuotaDoble ? 2 : 1);

        if (fila.numero === datos.cantidadCuotas) {
          fila.cuotaConstante =
            saldoInicial + fila.interes + fila.portes;
        }

        fila.amortizacion =
          fila.cuotaConstante - fila.portes - fila.interes;

        fila.saldoFinal = saldoInicial - fila.amortizacion;

        saldoInicial = fila.saldoFinal;
      }

      return {
        filas,
        sumaFactor1,
        sumaFactor2,
        cuotaAPagar
      };
    }

    function calcularPrima(datos) {
      const filas = crearCalendario(datos);

      let sumaFactor1 = 0;
      let sumaFactor2 = 0;

      for (const fila of filas) {
        const mes = fila.fechaPagoPrincipal.getMonth() + 1;

        fila.cuotaDoble =
          (mes === 7 && datos.cuotaJulioActiva) ||
          (mes === 12 && datos.cuotaDiciembreActiva);

        const denominador = Math.pow(
          1 + datos.teaTotalPrima,
          fila.diasAcumulados / 360
        );

        fila.factor1 = (fila.cuotaDoble ? 2 : 1) / denominador;
        fila.factor2 = 1 / denominador;

        sumaFactor1 += fila.factor1;
        sumaFactor2 += fila.factor2;
      }

      const cuotaPrima = datos.importe / sumaFactor1;

      let saldoInicial = datos.importe;
      let VPN = 0;

      for (const fila of filas) {
        fila.saldoInicial = saldoInicial;

        fila.interes = (
          Math.pow(1 + datos.tea, fila.dias / 360) - 1
        ) * saldoInicial;

        fila.segDesgravamen = (
          Math.pow(1 + datos.tasaSeguroPrima, fila.dias / 30) - 1
        ) * saldoInicial;

        fila.cuotaConstante =
          cuotaPrima * (fila.cuotaDoble ? 2 : 1);

        if (fila.numero === datos.cantidadCuotas) {
          fila.cuotaConstante =
            saldoInicial + fila.interes + fila.segDesgravamen;
        }

        fila.amortizacion =
          fila.cuotaConstante - fila.segDesgravamen - fila.interes;

        fila.saldoFinal = saldoInicial - fila.amortizacion;

        VPN += fila.segDesgravamen / Math.pow(
          1 + datos.tasaCostoCapital,
          fila.numero
        );

        saldoInicial = fila.saldoFinal;
      }

      return {
        filas,
        sumaFactor1,
        sumaFactor2,
        cuotaPrima,
        VPN
      };
    }

    function redondearACentimos(valor) {
      const [base, exponente = "0"] =
        Math.abs(valor).toString().split("e");

      const desplazado = Number(
        base + "e" + (Number(exponente) + 2)
      );

      return Math.sign(valor) * Math.round(desplazado);
    }

    function calcularVectorI(filasPrincipal, tcea) {
      const filas = filasPrincipal.map(fila => {
        const cuota =
          fila.amortizacion + fila.interes + fila.portes;

        const diasAcumulados = fila.diasAcumulados;

        const factor = Math.pow(
          1 + tcea,
          diasAcumulados / 360
        );

        // Redondeo real: los cálculos posteriores usan este valor.
        const calculoVACentimos = redondearACentimos(
          cuota / factor
        );

        const calculoVA = calculoVACentimos / 100;

        return {
          cuota,
          diasAcumulados,
          factor,
          calculoVA,
          calculoVACentimos
        };
      });

      const sumaVACentimos = filas.reduce(
        (suma, fila) => suma + fila.calculoVACentimos,
        0
      );

      return {
        filas,
        sumaVACentimos,
        sumaVA: sumaVACentimos / 100
      };
    }

    function calcularTCEA(importe, filasPrincipal) {
      const objetivoCentimos = redondearACentimos(importe);

      function evaluar(tcea) {
        const vector = calcularVectorI(filasPrincipal, tcea);

        return {
          tcea,
          ...vector,
          diferenciaCentimos:
            vector.sumaVACentimos - objetivoCentimos
        };
      }

      let inferior = 0;
      let superior = 1;

      let mejor = evaluar(inferior);

      if (mejor.diferenciaCentimos <= 0) {
        return mejor;
      }

      let extremoSuperior = evaluar(superior);

      for (
        let intento = 0;
        extremoSuperior.diferenciaCentimos > 0 && intento < 100;
        intento++
      ) {
        superior = superior * 2 + 1;
        extremoSuperior = evaluar(superior);
      }

      if (extremoSuperior.diferenciaCentimos > 0) {
        throw new Error(
          "No se pudo encontrar un intervalo para calcular la TCEA."
        );
      }

      if (
        Math.abs(extremoSuperior.diferenciaCentimos) <
        Math.abs(mejor.diferenciaCentimos)
      ) {
        mejor = extremoSuperior;
      }

      if (extremoSuperior.diferenciaCentimos === 0) {
        return extremoSuperior;
      }

      for (let intento = 0; intento < 200; intento++) {
        const medio = (inferior + superior) / 2;

        if (medio === inferior || medio === superior) {
          break;
        }

        const resultado = evaluar(medio);

        if (
          Math.abs(resultado.diferenciaCentimos) <
          Math.abs(mejor.diferenciaCentimos)
        ) {
          mejor = resultado;
        }

        if (resultado.diferenciaCentimos === 0) {
          return resultado;
        }

        if (resultado.diferenciaCentimos > 0) {
          inferior = medio;
        } else {
          superior = medio;
        }
      }

      return mejor;
    }

    function crearCalendario(datos) {
      const filas = [];

      let fechaPago = new Date(
        datos.fechaPrimerPago + "T00:00:00"
      );

      let fechaAnterior = new Date(
        datos.fechaDesembolso + "T00:00:00"
      );

      let diasAcumulados = 0;

      for (
        let numero = 1;
        numero <= datos.cantidadCuotas;
        numero++
      ) {
        const fechaPagoPrincipal = new Date(fechaPago);

        if (fechaPagoPrincipal.getDay() === 0) {
          fechaPagoPrincipal.setDate(
            fechaPagoPrincipal.getDate() + 1
          );
        }

        const dias = calcularDias(
          fechaAnterior,
          fechaPagoPrincipal
        );

        diasAcumulados += dias;

        filas.push({
          numero,
          fechaPago: new Date(fechaPago),
          fechaPagoPrincipal,
          dias,
          diasAcumulados
        });

        fechaAnterior = fechaPagoPrincipal;
        fechaPago = sumarUnMes(fechaPago);
      }

      return filas;
    }

    function sumarUnMes(fecha) {
      const siguienteFecha = new Date(fecha);
      const diaAnterior = siguienteFecha.getDate();

      siguienteFecha.setDate(1);
      siguienteFecha.setMonth(siguienteFecha.getMonth() + 1);

      const ultimoDiaDelMes = new Date(
        siguienteFecha.getFullYear(),
        siguienteFecha.getMonth() + 1,
        0
      ).getDate();

      siguienteFecha.setDate(
        Math.min(diaAnterior, ultimoDiaDelMes)
      );

      return siguienteFecha;
    }

    function calcularDias(fechaInicial, fechaFinal) {
      const inicio = Date.UTC(
        fechaInicial.getFullYear(),
        fechaInicial.getMonth(),
        fechaInicial.getDate()
      );

      const fin = Date.UTC(
        fechaFinal.getFullYear(),
        fechaFinal.getMonth(),
        fechaFinal.getDate()
      );

      return (fin - inicio) / (24 * 60 * 60 * 1000);
    }

    function formatearFecha(fecha) {
      const dia = String(fecha.getDate()).padStart(2, "0");
      const mes = String(fecha.getMonth() + 1).padStart(2, "0");
      const anio = String(fecha.getFullYear()).padStart(4, "0");

      return `${dia}/${mes}/${anio}`;
    }

    function crearTablaVectorI(resultadoVector) {

      return `

          </thead>

          <tbody>
            ${contenido}

          <tfoot>
            <tr>
              <th scope="row" colspan="3">Suma Cálculo VA</th>
              <td>${resultadoVector.sumaVA.toFixed(2)}</td>

    }

    function crearTabla(titulo, idTabla, filas, esPrincipal = false) {
      const contenido = filas.map(fila => `
        <tr>
          <td>${fila.numero}</td>
          <td>${formatearFecha(fila.fechaPago)}</td>
          <td>${formatearFecha(fila.fechaPagoPrincipal)}</td>
          <td>${fila.dias}</td>
          <td>${fila.diasAcumulados}</td>
          <td>${fila.saldoInicial.toFixed(2)}</td>
          <td>${fila.amortizacion.toFixed(2)}</td>
          <td>${fila.interes.toFixed(2)}</td>
          <td>${
            (esPrincipal ? fila.portes : fila.segDesgravamen)
              .toFixed(2)
          }</td>
          <td>${fila.cuotaConstante.toFixed(2)}</td>
          <td>${fila.saldoFinal.toFixed(2)}</td>
          <td>${fila.factor1.toFixed(2)}</td>
          <td>${fila.factor2.toFixed(2)}</td>
        </tr>
      `).join("");

      return `
        <h3>${titulo}</h3>

        <table id="${idTabla}">
          <thead>
            <tr>
              <th scope="col">N° de cuotas</th>
              <th scope="col">Fecha de pago</th>
              <th scope="col">Fecha de pago principal</th>
              <th scope="col">Días</th>
              <th scope="col">Días acumulados</th>
              <th scope="col">Saldo inicial</th>
              <th scope="col">Amortización</th>
              <th scope="col">Interés</th>
              <th scope="col">${
                esPrincipal ? "Portes" : "Seg de desgravamen"
              }</th>
              <th scope="col">Cuota constante</th>
              <th scope="col">Saldo final</th>
              <th scope="col">Factor1</th>
              <th scope="col">Factor 2</th>
            </tr>
          </thead>

          <tbody>
            ${contenido}
          </tbody>
        </table>
      `;
    }
