function simular() {
  const tipoSeguro = document.getElementById("seguro").value;

  // Valores expresados en porcentaje.
  const primas = {
    "vida": 3.7000,
    "desgravamen": 0.1995,
    "desgravamen-devolucion": 0.2594,
    "sin-seguro": 0,
    "endosado": 0
  };

  const seguroPrima = primas[tipoSeguro];
  const costoCapital = 0.2871;

  console.log(
    "Seguro prima:",
    seguroPrima === 0 ? "0%" : seguroPrima.toFixed(4) + "%"
  );

  console.log(
    "Costo capital:",
    costoCapital.toFixed(4) + "%"
  );

  const campoImporte = document.getElementById("importe");
  const campoPlazo = document.getElementById("plazo");
  const campoFecha = document.getElementById("primer-pago");
  const campoDesembolso = document.getElementById("desembolso");

  if (!campoImporte.reportValidity()) return;
  if (!campoPlazo.reportValidity()) return;
  if (!campoFecha.reportValidity()) return;
  if (!campoDesembolso.reportValidity()) return;

  const importe = Number(campoImporte.value);
  const cantidadCuotas = Number(campoPlazo.value);

  let fechaPago = new Date(campoFecha.value + "T00:00:00");

  const fechaDesembolso = new Date(
    campoDesembolso.value + "T00:00:00"
  );

  let fechaAnterior = fechaDesembolso;
  let diasAcumulados = 0;
  let saldoFinalAnterior = 0;
  let filas = "";

  for (let cuota = 1; cuota <= cantidadCuotas; cuota++) {
    const fechaPagoPrincipal = new Date(fechaPago);

    // Si cae domingo, pasa al lunes siguiente.
    if (fechaPagoPrincipal.getDay() === 0) {
      fechaPagoPrincipal.setDate(
        fechaPagoPrincipal.getDate() + 1
      );
    }

    const dias = calcularDias(
      fechaAnterior,
      fechaPagoPrincipal
    );

    diasAcumulados += dias;

    // Primera fila: importe.
    // Siguientes filas: saldo final de la fila anterior.
    const saldoInicial = cuota === 1
      ? importe
      : saldoFinalAnterior;

    // Columnas pendientes de fórmula: valor numérico cero.
    const interes = 0;
    const segDesgravamen = 0;
    const cuotaConstante = 0;
    const saldoFinal = 0;
    const factor1 = 0;
    const factor2 = 0;

    // Amortización de la fila actual.
    const amortizacion =
      cuotaConstante - segDesgravamen - interes;

    filas += `
      <tr>
        <td>${cuota}</td>
        <td>${formatearFecha(fechaPago)}</td>
        <td>${formatearFecha(fechaPagoPrincipal)}</td>
        <td>${dias}</td>
        <td>${diasAcumulados}</td>
        <td>${saldoInicial.toFixed(2)}</td>
        <td>${amortizacion.toFixed(2)}</td>
        <td>${interes.toFixed(2)}</td>
        <td>${segDesgravamen.toFixed(2)}</td>
        <td>${cuotaConstante.toFixed(2)}</td>
        <td>${saldoFinal.toFixed(2)}</td>
        <td>${factor1.toFixed(2)}</td>
        <td>${factor2.toFixed(2)}</td>
      </tr>
    `;

    // Valores de referencia para la siguiente fila.
    saldoFinalAnterior = saldoFinal;
    fechaAnterior = fechaPagoPrincipal;
    fechaPago = sumarUnMes(fechaPago);
  }

  document.getElementById("resultados").innerHTML = `
    <h3>Cálculo de prima</h3>

    <table>
      <thead>
        <tr>
          <th scope="col">N° de cuotas</th>
          <th scope="col">Fecha de pago</th>
          <th scope="col">Fecha de pago principal</th>
          <th scope="col">Días</th>
          <th scope="col">Días acumulados</th>
          <th scope="col">Saldo inicial</th>
          <th scope="col">Amortización</th>
          <th scope="col">Interés</th>
          <th scope="col">Seg de desgravamen</th>
          <th scope="col">Cuota constante</th>
          <th scope="col">Saldo final</th>
          <th scope="col">Factor1</th>
          <th scope="col">Factor 2</th>
        </tr>
      </thead>
      <tbody>
        ${filas}
      </tbody>
    </table>
  `;
}
