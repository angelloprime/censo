    const esRG = nombre =>
      nombre.trim().toUpperCase().startsWith('RG_');

    const escapar = valor =>
      String(valor).replace(/[&<>"']/g, c => ({
        '&':'&amp;',
        '<':'&lt;',
        '>':'&gt;',
        '"':'&quot;',
        "'":'&#39;'
      }[c]));

    const numero = valor => Number(valor).toFixed(2);

    const fechaTexto = iso =>
      iso.split('-').reverse().join('/');

    // Regla de tasa reducida:
    // Plazos de 1 a 35 meses = 79.99%.
    // Plazos de 36 meses o más = conserva la tasa registrada.
    const esPlazoCorto = plazo =>
      Number.isInteger(plazo) && plazo > 0 && plazo < 36;

    const ajustarTasaReducida = ficha =>
      esPlazoCorto(ficha.plazo)
        ? {...ficha,tasaReducida:79.99}
        : ficha;

    function hoyISO() {
      const fecha = new Date();

      return (
        `${fecha.getFullYear()}-` +
        `${String(fecha.getMonth()+1).padStart(2,'0')}-` +
        `${String(fecha.getDate()).padStart(2,'0')}`
      );
    }

    function fechaISO(texto) {
      const partes = /^(\d{2})\/(\d{2})\/(\d{4})$/.exec(texto);

      if (!partes) return null;

      const dia = Number(partes[1]);
      const mes = Number(partes[2]);
      const anio = Number(partes[3]);

      const fecha = new Date(0);
      fecha.setUTCFullYear(anio,mes-1,dia);

      if (
        anio < 1 ||
        fecha.getUTCFullYear() !== anio ||
        fecha.getUTCMonth() !== mes-1 ||
        fecha.getUTCDate() !== dia
      ) {
        return null;
      }

      return `${partes[3]}-${partes[2]}-${partes[1]}`;
    }

    function avisar(texto) {
      $('aviso').textContent = texto;
      $('aviso').hidden = false;
    }

    let persistencia = true;

    function fichaValida(f) {
      return (
        f &&
        typeof f.id === 'string' &&
        typeof f.nombre === 'string' &&
        f.nombre.trim() &&
        typeof f.dni === 'string' &&
        typeof f.telefono === 'string' &&
        Number.isFinite(f.tasa) &&
        f.tasa >= 0 &&
        (
          f.tasaReducida === null ||
          (
            Number.isFinite(f.tasaReducida) &&
            f.tasaReducida >= 0
          )
        ) &&
        Number.isFinite(f.importe) &&
        f.importe > 0 &&
        Number.isInteger(f.plazo) &&
        f.plazo > 0 &&
        CICLOS.includes(f.ciclo) &&
        typeof f.fechaDesembolso === 'string' &&
        fechaISO(fechaTexto(f.fechaDesembolso)) === f.fechaDesembolso &&
        ESTADOS.some(e => e.id === f.seccion)
      );
    }

    function cargar() {
      try {
        const guardado = localStorage.getItem(CLAVE);

        if (!guardado) return [];

        const datos = JSON.parse(guardado);

        if (
          !Array.isArray(datos) ||
          !datos.every(fichaValida) ||
          new Set(datos.map(f => f.id)).size !== datos.length
        ) {
          throw new Error();
        }

        return datos.map(ajustarTasaReducida);
      } catch {
        persistencia = false;

        avisar(
          'No se pudieron leer los datos guardados. ' +
          'Los cambios de esta sesión no se guardarán al cerrar.'
        );

        return [];
      }
    }

    let fichas = cargar();

    function guardar(nuevasFichas) {
      nuevasFichas = nuevasFichas.map(ajustarTasaReducida);

      if (persistencia) {
        try {
          localStorage.setItem(
            CLAVE,
            JSON.stringify(nuevasFichas)
          );
        } catch {
          persistencia = false;

          avisar(
            'El navegador no permitió guardar. ' +
            'Los cambios se conservarán solo durante esta sesión.'
          );
        }
      }

      fichas = nuevasFichas;
      renderizar();
    }

    function contadores(lista) {
      const rg = lista.filter(f => esRG(f.nombre)).length;

      return `
        <span class="contador">
          Total <b>${lista.length}</b>
        </span>
        <span class="contador contador-xl">
          XL <b>${lista.length-rg}</b>
        </span>
        <span class="contador contador-rg">
          RG <b>${rg}</b>
        </span>
      `;
    }

    function tarjeta(f) {
      const tipo = esRG(f.nombre) ? 'RG' : 'XL';

      const tasaTexto =
        numero(f.tasa) + '%' +
        (
          Number.isFinite(f.tasaReducida)
            ? ` → ${numero(f.tasaReducida)}%`
            : ''
        );

      const botones = [
        ['ver','eye','Ver'],
        ['editar','pencil-square','Editar'],
        ['mover','arrow-left-right','Mover'],
        ['simular','calculator','Simulación'],
        ['eliminar','trash3','Eliminar']
      ];

      return `
        <article class="ficha" data-id="${escapar(f.id)}">
          <div class="ficha-titulo">
            <h4>${escapar(f.nombre)}</h4>
            <span class="tipo contador-${tipo.toLowerCase()}">
              ${tipo}
            </span>
          </div>

          <dl class="ficha-datos">
            <div>
              <dt>DNI:</dt>
              <dd>${escapar(f.dni)}</dd>
            </div>
            <div>
              <dt>Tasa:</dt>
              <dd>${tasaTexto}</dd>
            </div>
            <div>
              <dt>Teléfono:</dt>
              <dd>${escapar(f.telefono)}</dd>
            </div>
            <div>
              <dt>Desembolso:</dt>
              <dd>${fechaTexto(f.fechaDesembolso)}</dd>
            </div>
          </dl>

          <div class="ficha-acciones">
            ${botones.map(([accion,icono,texto]) => `
              <button
                type="button"
                class="btn ${accion === 'eliminar' ? 'btn-eliminar' : ''}"
                data-accion="${accion}"
                ${
                  accion === 'eliminar'
                    ? 'aria-label="Eliminar ficha" title="Eliminar ficha"'
                    : ''
                }
              >
                <i class="bi bi-${icono}" aria-hidden="true"></i>
                ${accion === 'eliminar' ? '' : texto}
              </button>
            `).join('')}
          </div>
        </article>
      `;
    }

    function renderizar() {
      for (const grupo of ['porValidar','validados']) {
        const estados = ESTADOS.filter(e => e.grupo === grupo);

        $(grupo).innerHTML = estados.map(estado => {
          const lista = fichas.filter(f => f.seccion === estado.id);

          return `
            <section
              class="estado"
              id="${estado.id}"
              style="--acento:${estado.color}"
              aria-labelledby="titulo-${estado.id}"
            >
              <div class="estado-cabecera">
                <h3 id="titulo-${estado.id}">
                  <i class="bi bi-${estado.icono}" aria-hidden="true"></i>
                  ${estado.nombre}
                </h3>

                <div class="contadores">

  <script id="motorSimuladorXL">
    // Los importes se operan en céntimos para mantener
    // consistencia entre los valores mostrados y los saldos.
    const SimuladorXL = (() => {
      const DIA_MS = 86400000;

      function redondear(valor, decimales = 2) {
        if (!Number.isFinite(valor)) {
          throw new Error(
            'Los datos generan un resultado fuera de rango.'
          );
        }

        const desplazar = (n,lugares) => {
          const [base,exponente = '0'] = String(n).split('e');
          return Number(`${base}e${Number(exponente)+lugares}`);
        };

        const resultado =
          Math.sign(valor) *
          desplazar(
            Math.round(desplazar(Math.abs(valor),decimales)),
            -decimales
          );

        if (!Number.isFinite(resultado)) {
          throw new Error(
            'Los datos generan un resultado fuera de rango.'
          );
        }

        return resultado === 0 ? 0 : resultado;
      }

      function centimos(valor) {
        const c = Math.round(redondear(valor,2)*100);

        if (!Number.isSafeInteger(c)) {
          throw new Error(
            'El importe excede la precisión admitida.'
          );
        }

        return c;
      }

      function fechaUTC(anio,mes,dia) {
        const fecha = new Date(0);
        fecha.setUTCFullYear(anio,mes,dia);
        return fecha;
      }

      function aISO(fecha) {
        const anio = fecha.getUTCFullYear();

        if (
          !Number.isFinite(fecha.getTime()) ||
          anio < 1 ||
          anio > 9999
        ) {
          throw new Error(
            'El cronograma supera el rango de fechas admitido.'
          );
        }

        return (
          `${String(anio).padStart(4,'0')}-` +
          `${String(fecha.getUTCMonth()+1).padStart(2,'0')}-` +
          `${String(fecha.getUTCDate()).padStart(2,'0')}`
        );
      }

      // TIR periódica con flujos mensuales.
      function tirPeriodica(importe,pagos,estimacion) {
        if (
          !(importe > 0) ||
          pagos.some(p => !Number.isFinite(p) || p < 0) ||
          !pagos.some(p => p > 0)
        ) {
          return null;
        }

        const valorActual = x => {
          let suma = -importe;

          for (let i=0; i<pagos.length; i++) {
            if (pagos[i] !== 0) {
              suma += pagos[i]*Math.exp(-(i+1)*x);
            }
          }

          return suma;
        };

        let inferior = -1;
        let superior = Math.max(1,Math.log1p(estimacion));
        let limite = 0;

        while (valorActual(inferior) < 0 && limite++ < 20) {
          inferior *= 2;
        }

        limite = 0;

        while (valorActual(superior) > 0 && limite++ < 20) {
          superior *= 2;
        }

        if (!(
          valorActual(inferior) >= 0 &&
          valorActual(superior) <= 0
        )) {
          return null;
        }

        let centro = Math.min(
          superior,
          Math.max(inferior,Math.log1p(estimacion))
        );

        for (let i=0; i<200; i++) {
          const valor = valorActual(centro);

          if (
            Math.abs(valor) <= Math.max(importe*1e-13,1e-12)
          ) {
            break;
          }

          if (valor > 0) {
            inferior = centro;
          } else {
            superior = centro;
          }

          centro = (inferior+superior)/2;
        }

        const tir = Math.expm1(centro);

        return Number.isFinite(tir) && tir > -1 ? tir : null;
      }

      function calcular({importe,plazo,fecha,ciclo,tea,seguro}) {
        if (
          !Number.isFinite(importe) ||
          importe <= 0 ||
          !Number.isSafeInteger(plazo) ||
          plazo <= 0 ||
          !Number.isFinite(tea) ||
          tea < 0 ||
          ![4,6,9,11,13,16,18,20,22,24,26].includes(ciclo) ||
          ![0.004,0.003].includes(seguro)
        ) {
          throw new Error(
            'Revisa el importe, plazo, TEA, ciclo y seguro.'
          );
        }

        if (plazo > 10000) {
          throw new Error(
            'Ingresa un plazo de hasta 10000 meses.'
          );
        }

        if (
          typeof fecha !== 'string' ||
          !/^\d{4}-\d{2}-\d{2}$/.test(fecha)
        ) {
          throw new Error('Ingresa una fecha válida.');
        }

        const [anio,mes,dia] = fecha.split('-').map(Number);
        const desembolso = fechaUTC(anio,mes-1,dia);

        if (aISO(desembolso) !== fecha) {
          throw new Error('Ingresa una fecha válida.');
        }

        const importeCt = centimos(importe);

        if (importeCt <= 0) {
          throw new Error('El importe mínimo es 0.01.');
        }

        const filas = [];

        // Si el día del desembolso es igual o menor al ciclo:
        // se utiliza el mismo mes. En caso contrario, el siguiente.
        const primerMes = mes-1+(dia <= ciclo ? 0 : 1);

        let anterior = desembolso;
        let diasAcum = 0;

        for (let i=0; i<plazo; i++) {
          const facturacion = fechaUTC(anio,primerMes+i,ciclo);

          // Vencimiento: ciclo de facturación - 3 días + 1 mes.
          const vencimiento = fechaUTC(
            facturacion.getUTCFullYear(),
            facturacion.getUTCMonth()+1,
            ciclo-3
          );

          const dias =
            Math.round((vencimiento-anterior)/DIA_MS) +
            (i === 0 ? 1 : 0);

          diasAcum += dias;

          const factor = redondear(
            1/Math.pow(1+tea,diasAcum/360),
            9
          );

          filas.push({
            numero:i+1,
            facturacion:aISO(facturacion),
            vencimiento:aISO(vencimiento),
            dias,
            diasAcum,
            factor
          });

          anterior = vencimiento;
        }

        // Cada factor se redondea a 9 decimales.
        // La suma de los factores no se redondea.
        const sumaFactor = filas.reduce(
          (suma,f) => suma+f.factor,
          0
        );

        if (!(sumaFactor > 0)) {
          throw new Error(
            'No se puede calcular la cuota con estos datos.'
          );
        }

        const cuotaCt = centimos((importeCt/100)/sumaFactor);

        let pendienteCt = importeCt;
        let amortAcumCt = 0;
        let amortPrincipalAcumCt = 0;
        let totalCt = 0;

        for (let i=0; i<filas.length; i++) {
          const f = filas[i];
          const ultima = i === filas.length-1;

          let interesYCt = ultima
            ? 0
            : centimos(
                (pendienteCt/100) *
                Math.expm1(Math.log1p(tea)*f.dias/360)
              );

          const amortizacionCt = ultima
            ? pendienteCt
            : cuotaCt-interesYCt;

          if (ultima) {
            interesYCt = cuotaCt-amortizacionCt;
          }

          const saldoCt = pendienteCt-amortizacionCt;

          amortAcumCt += amortizacionCt;

          const amortPrincipalCt = amortAcumCt <= 0
            ? 100
            : amortAcumCt-amortPrincipalAcumCt;

          amortPrincipalAcumCt += amortPrincipalCt;

          const interesCt = cuotaCt-amortPrincipalCt;

          const seguroCt = centimos(
            Math.min(30,(pendienteCt/100)*seguro)
          );

          const cuotaSeguroCt = cuotaCt+seguroCt;

          totalCt += cuotaSeguroCt;

          if (![
            saldoCt,
            amortAcumCt,
            amortPrincipalAcumCt,
            totalCt
          ].every(Number.isSafeInteger)) {
            throw new Error(
              'Los resultados exceden la precisión admitida.'
            );
          }

          Object.assign(f,{
            principal:pendienteCt/100,
            amortizacion:amortizacionCt/100,
            interesY:interesYCt/100,
            cuotaY:cuotaCt/100,
            saldo:saldoCt/100,
            amortPrincipal:amortPrincipalCt/100,
            interes:interesCt/100,
            cuota:cuotaCt/100,
            seguro:seguroCt/100,
            cuotaSeguro:cuotaSeguroCt/100
          });

          pendienteCt = saldoCt;
        }

        const tem = Math.expm1(Math.log1p(tea)/12);

        const flujos = [
          -(importeCt/100),
          ...filas.map(f => f.cuotaSeguro)
        ];

        const tir = tirPeriodica(
          importeCt/100,
          flujos.slice(1),
          tem
        );

        const anual = tir === null
          ? null
          : Math.expm1(12*Math.log1p(tir));

        const tcea = Number.isFinite(anual) ? anual : null;

        const tcem = tcea === null
          ? null
          : Math.expm1(Math.log1p(tcea)/12);

        return {
          filas,
          sumaFactor,
          cuota:cuotaCt/100,
          tem,
          tcea,
          tcem,
          total:totalCt/100,
          flujos
        };
      }

      return {
        calcular,
        redondear,
        tirPeriodica
      };
    })();
  </script>

          error.message || 'No se pudo calcular la simulación.';

        $('simError').hidden = false;
      }
    });
  </script>
